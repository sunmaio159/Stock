### 3.1.1 采集调度器伪代码

```java
/**
 * 采集调度器 —— 基于 ThreadPoolTaskScheduler + AtomicBoolean 实现
 * 功能：定时触发采集任务、限流、熔断保护、增量游标管理
 */
@Component
@Slf4j
public class CrawlScheduler {

    @Autowired private CrawlTaskMapper crawlTaskMapper;
    @Autowired private CrawlTaskRunMapper crawlTaskRunMapper;
    @Autowired private CrawlSourceHealthMapper healthMapper;
    @Autowired private CrawlExecutor crawlExecutor;
    @Autowired private RedisTemplate<String, String> redisTemplate;

    /** 调度线程池：8线程，足以覆盖东财股吧/行情/新闻等多源并发 */
    private final ThreadPoolTaskScheduler scheduler = new ThreadPoolTaskScheduler();
    /** 全局运行开关 —— AtomicBoolean 保证多线程可见性，可热启停 */
    private final AtomicBoolean running = new AtomicBoolean(false);
    /** 每个任务的运行状态锁，防止同一任务并发重入 */
    private final ConcurrentMap<Long, AtomicBoolean> taskLocks = new ConcurrentHashMap<>();

    @PostConstruct
    public void init() {
        scheduler.setPoolSize(8);
        scheduler.setThreadNamePrefix("crawl-sched-");
        scheduler.setWaitForTasksToCompleteOnShutdown(true);
        scheduler.setAwaitTerminationSeconds(30);
        scheduler.initialize();
        // 启动主扫描循环，每30秒扫描一次到期任务
        scheduler.scheduleAtFixedRate(this::scanAndDispatch, Duration.ofSeconds(30));
        log.info("CrawlScheduler initialized, poolSize=8");
    }

    /**
     * 扫描到期任务并分发执行
     */
    private void scanAndDispatch() {
        if (!running.get()) {
            return;
        }
        // 从DB查询所有启用且到期的任务
        // next_run_time <= now() AND status = 1
        List<CrawlTask> dueTasks = crawlTaskMapper.selectDueTasks(LocalDateTime.now());
        for (CrawlTask task : dueTasks) {
            // 每个任务用独立的 AtomicBoolean 防重入
            AtomicBoolean taskLock = taskLocks.computeIfAbsent(task.getId(), k -> new AtomicBoolean(false));
            if (!taskLock.compareAndSet(false, true)) {
                log.debug("Task {} already running, skip", task.getId());
                continue;
            }
            // 提交到线程池异步执行
            scheduler.submit(() -> {
                try {
                    executeTaskWithGuard(task);
                } catch (Exception e) {
                    log.error("Task {} failed: {}", task.getId(), e.getMessage(), e);
                } finally {
                    taskLock.set(false);
                    // 更新下次执行时间
                    updateNextRunTime(task);
                }
            });
        }
    }

    /**
     * 带限流+熔断保护的任务执行
     */
    private void executeTaskWithGuard(CrawlTask task) {
        // 1. 检查源健康度 —— 熔断判断
        CrawlSourceHealth health = healthMapper.selectBySourceCode(task.getSourceCode());
        if (health != null && health.isCircuitBreak()) {
            log.warn("Source {} circuit breaker OPEN, skip task {}", task.getSourceCode(), task.getId());
            return;
        }
        // 2. 检查连续失败次数 >= 5 则熔断
        if (health != null && health.getConsecutiveFailures() >= 5) {
            health.setCircuitBreak(true);
            health.setCircuitBreakUntil(LocalDateTime.now().plusMinutes(30));
            healthMapper.updateById(health);
            log.warn("Source {} triggered circuit breaker (consecutive failures >= 5)", task.getSourceCode());
            return;
        }
        // 3. Redis限流 —— 每源QPS控制
        String rateLimitKey = "crawl:rate:" + task.getSourceCode();
        String current = redisTemplate.opsForValue().get(rateLimitKey);
        int maxQps = task.getMaxQps() != null ? task.getMaxQps() : 5; // 默认5 QPS
        if (current != null && Integer.parseInt(current) >= maxQps) {
            log.debug("Source {} rate limited, skip", task.getSourceCode());
            return;
        }
        // 4. 增量游标 —— 从上次游标继续采集
        String cursorKey = "crawl:cursor:" + task.getId();
        String lastCursor = redisTemplate.opsForValue().get(cursorKey);
        task.setIncrementalCursor(lastCursor); // 注入增量游标（如最后一条帖子ID/时间戳）

        // 5. 创建运行实例
        CrawlTaskRun run = new CrawlTaskRun();
        run.setTaskId(task.getId());
        run.setSourceCode(task.getSourceCode());
        run.setStatus("RUNNING");
        run.setStartTime(LocalDateTime.now());
        run.setCursor(lastCursor);
        crawlTaskRunMapper.insert(run);

        // 6. 执行采集
        CrawlResult result;
        try {
            result = crawlExecutor.execute(task);
            // 成功：重置失败计数，更新游标
            if (health != null) {
                health.setConsecutiveFailures(0);
                health.setCircuitBreak(false);
                health.setLastSuccessTime(LocalDateTime.now());
                healthMapper.updateById(health);
            }
            if (result.getNewCursor() != null) {
                redisTemplate.opsForValue().set(cursorKey, result.getNewCursor());
            }
            run.setStatus("SUCCESS");
            run.setRecordCount(result.getRecordCount());
        } catch (Exception e) {
            run.setStatus("FAILED");
            run.setErrorMsg(e.getMessage());
            if (health != null) {
                health.setConsecutiveFailures(health.getConsecutiveFailures() + 1);
                healthMapper.updateById(health);
            }
            throw e;
        } finally {
            run.setEndTime(LocalDateTime.now());
            crawlTaskRunMapper.updateById(run);
        }
    }

    /**
     * 根据cron表达式更新下次执行时间
     */
    private void updateNextRunTime(CrawlTask task) {
        if (task.getCronExpr() != null) {
            CronExpression cron = CronExpression.parse(task.getCronExpr());
            LocalDateTime next = cron.next(LocalDateTime.now());
            task.setNextRunTime(next);
        } else {
            // 固定间隔（秒）
            int interval = task.getIntervalSeconds() != null ? task.getIntervalSeconds() : 300;
            task.setNextRunTime(LocalDateTime.now().plusSeconds(interval));
        }
        crawlTaskMapper.updateNextRunTime(task.getId(), task.getNextRunTime());
    }

    /** 热启动 */
    public void start() {
        running.set(true);
        log.info("CrawlScheduler started");
    }

    /** 热停止 */
    public void stop() {
        running.set(false);
        log.info("CrawlScheduler stopped");
    }

    @PreDestroy
    public void destroy() {
        running.set(false);
        scheduler.shutdown();
    }
}
```

### 3.1.2 Adapter插件化架构伪代码

```java
/**
 * 采集适配器接口 —— 所有数据源的统一入口
 * 实现"插件化"：新增数据源只需实现此接口并注册到 AdapterRegistry
 */
public interface CrawlAdapter {

    /** 数据源唯一编码，对应 crawl_source.source_code */
    String getSourceCode();

    /** 数据源类型：GUBA / QUOTE / NEWS / ANNOUNCEMENT / RESEARCH */
    String getDataType();

    /**
     * 执行采集
     * @param params 采集参数（板块代码、股票代码、页码、游标等）
     * @param context 采集上下文（Cookie、UA、代理、限流器等）
     * @return 采集到的原始记录列表 + 新游标
     */
    CrawlResult fetch(CrawlParams params, CrawlContext context);

    /**
     * 解析原始HTML/JSON为NormalizedDoc列表
     * @param rawRecords 原始记录
     * @return 归一化文档
     */
    List<NormalizedDoc> parse(List<RawRecord> rawRecords);

    /** Schema校验 —— 检查解析后的文档是否符合预期字段 */
    List<FieldDrift> validateSchema(List<NormalizedDoc> docs);
}

/**
 * Adapter注册中心 —— Spring自动扫描所有CrawlAdapter实现并注册
 */
@Component
@Slf4j
public class AdapterRegistry {

    private final Map<String, CrawlAdapter> adapterMap = new ConcurrentHashMap<>();

    @Autowired
    public AdapterRegistry(List<CrawlAdapter> adapters) {
        for (CrawlAdapter adapter : adapters) {
            adapterMap.put(adapter.getSourceCode() + ":" + adapter.getDataType(), adapter);
            log.info("Registered adapter: {}:{}", adapter.getSourceCode(), adapter.getDataType());
        }
    }

    public CrawlAdapter getAdapter(String sourceCode, String dataType) {
        String key = sourceCode + ":" + dataType;
        CrawlAdapter adapter = adapterMap.get(key);
        if (adapter == null) {
            throw new IllegalArgumentException("No adapter found for " + key);
        }
        return adapter;
    }

    /** 获取某数据类型可用的所有数据源（用于多源降级路由） */
    public List<CrawlAdapter> getAdaptersByDataType(String dataType) {
        return adapterMap.values().stream()
            .filter(a -> a.getDataType().equals(dataType))
            .collect(Collectors.toList());
    }
}

/**
 * 东财股吧 Adapter 实现示例
 * 数据源：东方财富股吧帖子列表API
 */
@Component
@Slf4j
public class EastmoneyGubaAdapter implements CrawlAdapter {

    private static final String BASE_URL = "https://guba.eastmoney.com/list,%s_%d.html";
    @Autowired private HttpClient httpClient; // Apache HttpClient5
    @Autowired private CrawlSourceHealthMapper healthMapper;

    @Override
    public String getSourceCode() { return "eastmoney"; }

    @Override
    public String getDataType() { return "GUBA"; }

    @Override
    public CrawlResult fetch(CrawlParams params, CrawlContext context) {
        String stockCode = params.getString("stockCode");  // e.g. "600519"
        int pageNum = params.getInt("pageNum", 1);
        String cursor = params.getString("cursor");        // 增量游标：最后帖子ID

        String url = String.format(BASE_URL, stockCode, pageNum);
        // 构建HTTP请求（带UA/Header）
        HttpRequest request = HttpRequest.newBuilder()
            .uri(URI.create(url))
            .header("User-Agent", context.getUA("Mozilla/5.0 (Windows NT 10.0; Win64; x64)"))
            .header("Referer", "https://guba.eastmoney.com/")
            .timeout(Duration.ofSeconds(15))
            .GET()
            .build();

        HttpResponse<String> response;
        try {
            // 限流等待
            context.acquireRatePermit();
            response = httpClient.send(request, HttpResponse.BodyHandlers.ofString());
        } catch (Exception e) {
            throw new CrawlException("Eastmoney guba fetch failed: " + e.getMessage(), e);
        }

        if (response.statusCode() != 200) {
            throw new CrawlException("Eastmoney guba HTTP " + response.statusCode());
        }

        String html = response.body();
        // 保存原始记录
        RawRecord rawRecord = new RawRecord();
        rawRecord.setSourceCode("eastmoney");
        rawRecord.setDataType("GUBA");
        rawRecord.setStockCode(stockCode);
        rawRecord.setPageNum(pageNum);
        rawRecord.setRawHtml(html);
        rawRecord.setFetchTime(LocalDateTime.now());

        // 提取新游标（最后帖子发布时间或ID）
        String newCursor = extractLastPostId(html);

        CrawlResult result = new CrawlResult();
        result.addRawRecord(rawRecord);
        result.setNewCursor(newCursor);
        return result;
    }

    @Override
    public List<NormalizedDoc> parse(List<RawRecord> rawRecords) {
        List<NormalizedDoc> docs = new ArrayList<>();
        for (RawRecord raw : rawRecords) {
            Document doc = Jsoup.parse(raw.getRawHtml());
            // 股吧帖子列表选择器
            Elements posts = doc.select("div.articleh");
            for (Element post : posts) {
                NormalizedDoc nd = new NormalizedDoc();
                nd.setSourceCode("eastmoney");
                nd.setDataType("GUBA");
                nd.setStockCode(raw.getStockCode());

                // 帖子标题
                Element titleEl = post.selectFirst("span.l3 a");
                if (titleEl != null) {
                    nd.setTitle(titleEl.text().trim());
                    nd.setUrl("https://guba.eastmoney.com" + titleEl.attr("href"));
                }
                // 作者
                Element authorEl = post.selectFirst("span.l4 a");
                if (authorEl != null) {
                    nd.setAuthor(authorEl.text().trim());
                }
                // 发布时间
                Element timeEl = post.selectFirst("span.l5");
                if (timeEl != null) {
                    String timeStr = timeEl.text().trim();
                    nd.setPublishTime(parseGubaTime(timeStr));
                }
                // 阅读数/评论数
                Element readEl = post.selectFirst("span.l1");
                Element replyEl = post.selectFirst("span.l2");
                if (readEl != null) nd.setViewCount(parseIntSafe(readEl.text()));
                if (replyEl != null) nd.setReplyCount(parseIntSafe(replyEl.text()));

                nd.setRawRecordId(raw.getId());
                docs.add(nd);
            }
        }
        return docs;
    }

    @Override
    public List<FieldDrift> validateSchema(List<NormalizedDoc> docs) {
        List<FieldDrift> drifts = new ArrayList<>();
        int missingTitle = 0, missingAuthor = 0, missingTime = 0;
        for (NormalizedDoc d : docs) {
            if (d.getTitle() == null) missingTitle++;
            if (d.getAuthor() == null) missingAuthor++;
            if (d.getPublishTime() == null) missingTime++;
        }
        int total = docs.size();
        if (total == 0) return drifts;

        double titleRate = (double) missingTitle / total;
        double authorRate = (double) missingAuthor / total;
        double timeRate = (double) missingTime / total;

        if (titleRate > 0.3) {
            drifts.add(new FieldDrift("title", titleRate, 0.3, "title missing rate exceeds 30%"));
        }
        if (authorRate > 0.3) {
            drifts.add(new FieldDrift("author", authorRate, 0.3, "author missing rate exceeds 30%"));
        }
        if (timeRate > 0.3) {
            drifts.add(new FieldDrift("publishTime", timeRate, 0.3, "publishTime missing rate exceeds 30%"));
        }
        return drifts;
    }

    private LocalDateTime parseGubaTime(String timeStr) {
        // 处理 "09-30 12:00" 或 "2024-09-30 12:00:00" 等格式
        if (timeStr.contains("-") && timeStr.length() <= 11) {
            return LocalDateTime.parse(LocalDate.now().getYear() + "-" + timeStr + ":00",
                DateTimeFormatter.ofPattern("yyyy-MM-dd HH:mm:ss"));
        }
        return LocalDateTime.parse(timeStr, DateTimeFormatter.ofPattern("yyyy-MM-dd HH:mm:ss"));
    }

    private String extractLastPostId(String html) {
        Document doc = Jsoup.parse(html);
        Element lastPost = doc.selectLast("span.l5");
        return lastPost != null ? lastPost.text().trim() : null;
    }

    private int parseIntSafe(String s) {
        try { return Integer.parseInt(s.trim()); } catch (Exception e) { return 0; }
    }
}

/**
 * 行情数据 Adapter 实现示例
 * 数据源：东财行情API（K线日线/分钟线）

@Component
@Slfj
public class QuoteAdapter implements CrawlAdapter {

    // 东财K线API: http://push2his.eastmoney.com/api/qt/stock/kline/get
    private static final String KLINE_URL =
        "http://push2his.eastmoney.com/api/qt/stock/kline/get" +
        "?secid=%s&fields1=f1,f2,f3,f4,f5,f6&fields2=f51,f52,f53,f54,f55,f56,f57,f58&klt=%d&fqt=1";

    @Autowired private HttpClient httpClient;

    @Override
    public String getSourceCode() { return "eastmoney"; }

    @Override
    public String getDataType() { return "QUOTE"; }

    @Override
    public CrawlResult fetch(CrawlParams params, CrawlContext context) {
        String secid = params.getString("secid");     // e.g. "1.600519"(沪) / "0.000001"(深)
        int klt = params.getInt("klt", 101);           // 101=日线 5=5分钟 15=15分钟
        String url = String.format(KLINE_URL, secid, klt);

        HttpRequest request = HttpRequest.newBuilder()
            .uri(URI.create(url))
            .header("User-Agent", context.getUA("Mozilla/5.0"))
            .header("Referer", "https://quote.eastmoney.com/")
            .timeout(Duration.ofSeconds(10))
            .GET()
            .build();

        HttpResponse<String> response;
        try {
            context.acquireRatePermit();
            response = httpClient.send(request, HttpResponse.BodyHandlers.ofString());
        } catch (Exception e) {
            throw new CrawlException("Quote fetch failed", e);
        }

        if (response.statusCode() != 200) {
            throw new CrawlException("Quote HTTP " + response.statusCode());
        }

        RawRecord raw = new RawRecord();
        raw.setSourceCode("eastmoney");
        raw.setDataType("QUOTE");
        raw.setRawJson(response.body());
        raw.setFetchTime(LocalDateTime.now());
        raw.setStockCode(params.getString("stockCode"));

        CrawlResult result = new CrawlResult();
        result.addRawRecord(raw);
        return result;
    }

    @Override
    @SuppressWarnings("unchecked")
    public List<NormalizedDoc> parse(List<RawRecord> rawRecords) {
        List<NormalizedDoc> docs = new ArrayList<>();
        ObjectMapper mapper = new ObjectMapper();
        for (RawRecord raw : rawRecords) {
            try {
                JsonNode root = mapper.readTree(raw.getRawJson());
                JsonNode klines = root.path("data").path("klines");
                if (klines.isMissingNode() || !klines.isArray()) continue;

                for (JsonNode kline : klines) {
                    // 格式：日期,开盘,收盘,最高,最低,成交量,成交额,振幅
                    String[] parts = kline.asText().split(",");
                    NormalizedDoc doc = new NormalizedDoc();
                    doc.setSourceCode("eastmoney");
                    doc.setDataType("QUOTE");
                    doc.setStockCode(raw.getStockCode());
                    doc.setTradeDate(parts[0]);
                    doc.setOpen(new BigDecimal(parts[1]));
                    doc.setClose(new BigDecimal(parts[2]));
                    doc.setHigh(new BigDecimal(parts[3]));
                    doc.setLow(new BigDecimal(parts[4]));
                    doc.setVolume(new BigDecimal(parts[5]));
                    doc.setAmount(new BigDecimal(parts[6]));
                    docs.add(doc);
                }
            } catch (Exception e) {
                log.error("Parse quote failed: {}", e.getMessage());
            }
        }
        return docs;
    }

    @Override
    public List<FieldDrift> validateSchema(List<NormalizedDoc> docs) {
        List<FieldDrift> drifts = new ArrayList<>();
        int missingClose = 0;
        for (NormalizedDoc d : docs) {
            if (d.getClose() == null) missingClose++;
        }
        if (!docs.isEmpty() && (double) missingClose / docs.size() > 0.3) {
            drifts.add(new FieldDrift("close", (double) missingClose / docs.size(), 0.3, "close price missing"));
        }
        return drifts;
    }
}
```

### 3.1.3 RawRecord落库与重放机制伪代码

```java
/**
 * RawRecord落库与重放机制
 * 1. 所有采集的原始HTML/JSON先存 crawl_raw_record 表（保留30天）
 * 2. 支持失败重放：从DB读取原始记录，重新走解析→治理流程
 */
@Service
@Slf4j
public class RawRecordService {

    @Autowired private CrawlRawRecordMapper rawRecordMapper;
    @Autowired private AdapterRegistry adapterRegistry;
    @Autowired private GovernancePipelineService governancePipeline;

    private static final int RETENTION_DAYS = 30;

    /**
     * 批量落库原始记录
     */
    @Transactional
    public void batchSave(List<RawRecord> records) {
        if (records == null || records.isEmpty()) return;
        for (RawRecord record : records) {
            record.setStatus("PENDING"); // PENDING / PARSED / FAILED
            record.setFetchTime(LocalDateTime.now());
            rawRecordMapper.insert(record);
        }
        log.info("Saved {} raw records", records.size());
    }

    /**
     * 重放原始记录 —— 当Adapter解析逻辑更新后，重新解析历史数据
     * @param taskId 采集任务ID
     * @param startTime 重放起始时间
     * @param endTime 重放结束时间
     */
    public void replay(Long taskId, LocalDateTime startTime, LocalDateTime endTime) {
        log.info("Start replay: taskId={}, range=[{}, {}]", taskId, startTime, endTime);

        // 分页查询历史原始记录
        int pageNum = 1;
        int pageSize = 200;
        while (true) {
            List<RawRecord> batch = rawRecordMapper.selectByTaskAndTimeRange(
                taskId, startTime, endTime, (pageNum - 1) * pageSize, pageSize);
            if (batch.isEmpty()) break;

            for (RawRecord raw : batch) {
                try {
                    CrawlAdapter adapter = adapterRegistry.getAdapter(raw.getSourceCode(), raw.getDataType());
                    // 重新解析
                    List<NormalizedDoc> docs = adapter.parse(List.of(raw));
                    if (!docs.isEmpty()) {
                        // 重新进入治理流水线
                        governancePipeline.processBatch(docs);
                    }
                    raw.setStatus("PARSED");
                } catch (Exception e) {
                    raw.setStatus("FAILED");
                    raw.setErrorMsg(e.getMessage());
                    log.error("Replay record {} failed: {}", raw.getId(), e.getMessage());
                }
                rawRecordMapper.updateStatus(raw.getId(), raw.getStatus(), raw.getErrorMsg());
            }
            pageNum++;
        }
        log.info("Replay completed: taskId={}", taskId);
    }

    /**
     * 清理过期原始记录（保留30天）—— 定时任务每天凌晨执行
     */
    @Scheduled(cron = "0 0 2 * * ?")
    public void cleanupExpiredRecords() {
        LocalDateTime cutoff = LocalDateTime.now().minusDays(RETENTION_DAYS);
        int deleted = rawRecordMapper.deleteBeforeTime(cutoff);
        log.info("Cleaned up {} expired raw records before {}", deleted, cutoff);
    }
}
```

### 3.1.4 字段漂移检测伪代码

```java
/**
 * 字段漂移检测 —— 监控数据源结构变化
 * 当某字段缺失率超过阈值（如30%）时触发告警
 * 防止数据源页面改版导致批量解析失败却无感知
 */
@Service
@Slf4j
public class FieldDriftDetector {

    @Autowired private RedisTemplate<String, String> redisTemplate;
    @Autowired private AlertEventService alertEventService;

    private static final double DRIFT_THRESHOLD = 0.3;       // 缺字段率阈值30%
    private static final double ALERT_THRESHOLD = 0.5;       // 告警阈值50%

    /**
     * 检测并记录字段漂移
     * @param sourceCode 数据源编码
     * @param dataType 数据类型
     * @param drifts 漂移列表
     */
    public void detect(String sourceCode, String dataType, List<FieldDrift> drifts) {
        if (drifts == null || drifts.isEmpty()) return;

        for (FieldDrift drift : drifts) {
            String key = String.format("drift:%s:%s:%s", sourceCode, dataType, drift.getFieldName());

            // 记录漂移历史到Redis（最近100次采样）
            redisTemplate.opsForList().rightPush(key, String.valueOf(drift.getMissingRate()));
            redisTemplate.opsForList().trim(key, -100, -1); // 只保留最近100次
            redisTemplate.expire(key, Duration.ofDays(7));

            if (drift.getMissingRate() > ALERT_THRESHOLD) {
                // 超告警阈值 —— 触发告警
                String alertMsg = String.format(
                    "字段漂移告警: source=%s, type=%s, field=%s, missingRate=%.2f(阈值%.2f)",
                    sourceCode, dataType, drift.getFieldName(),
                    drift.getMissingRate(), ALERT_THRESHOLD);
                log.warn(alertMsg);

                // 防止重复告警：同一漂移1小时内只报一次
                String dedupKey = "drift:alert:" + key;
                Boolean firstTime = redisTemplate.opsForValue()
                    .setIfAbsent(dedupKey, "1", Duration.ofHours(1));
                if (Boolean.TRUE.equals(firstTime)) {
                    alertEventService.raiseAlert(
                        AlertEvent.builder()
                            .level("WARN")
                            .category("DATA_DRIFT")
                            .title("字段漂移: " + sourceCode + ":" + drift.getFieldName())
                            .content(alertMsg)
                            .scope("SYSTEM")
                            .build());
                }
            } else if (drift.getMissingRate() > DRIFT_THRESHOLD) {
                log.warn("Field drift detected (below alert threshold): {}", drift);
            }
        }

    /**
     * 获取漂移趋势 —— 供管理页面查看
     */
    public Map<String, List<Double>> getDriftHistory(String sourceCode, String dataType) {

                    String fieldName = key.substring(key.lastIndexOf(':') + 1);
                    result.put(fieldName, values.stream()
                        .map(Double::parseDouble)
                        .collect(Collectors.toList()));
                }
            }
```

```java
/**
 * DedupWorker —— SimHash64 去重
 * 1. 对清洗后内容计算SimHash64指纹
 * 2. 与Redis滑窗内已有指纹比对，汉明距离≤3判重
 */
@Component
@Slf4j
public class DedupWorker {

    @Autowired private RedisTemplate<String, String> redisTemplate;

    private static final int SLIDING_WINDOW_HOURS = 24;  // 滑窗24小时
    private static final int HAMMING_THRESHOLD = 3;      // 汉明距离阈值
    private static final int MAX_SIMHASH_INDEX = 5000;   // 每个股票索引上限

    /**
     * SimHash64 计算 —— 分词→hash→加权→合并→降维
     */
    public long computeSimHash(String text) {
        if (text == null || text.isEmpty()) return 0L;

        // 1. 分词（简单按字符二元组 + 空格分词）
        List<String> tokens = tokenize(text);

        // 2. 每个token hash后，按位加权
        int[] bits = new int[64];
        Map<Long, Integer> tokenWeights = new HashMap<>();

        for (String token : tokens) {
            long hash = murmurHash3Long(token);
            tokenWeights.merge(hash, 1, Integer::sum);
        }

        for (Map.Entry<Long, Integer> entry : tokenWeights.entrySet()) {
            long hash = entry.getKey();
            int weight = entry.getValue();
            for (int i = 0; i < 64; i++) {
                if ((hash & (1L << i)) != 0) {
                    bits[i] += weight;
                } else {
                    bits[i] -= weight;
                }
            }
        }

        // 3. 降维：>0 取1，<=0 取0
        long simhash = 0L;
        for (int i = 0; i < 64; i++) {
            if (bits[i] > 0) {
                simhash |= (1L << i);
            }
        }
        return simhash;
    }

    /**
     * 去重判断 —— 判断是否为重复内容
     * @param doc 待判断文档
     * @return true=重复(丢弃), false=非重复(保留)
     */
    public boolean isDuplicate(NormalizedDoc doc) {
        long simhash = computeSimHash(doc.getTitle() + " " + doc.getContent());
        doc.setSimHash(simhash);

        String stockCode = doc.getStockCode();
        String indexKey = String.format("simhash:index:%s", stockCode);

        // 从Redis加载该股票的SimHash索引
        Set<String> existingHashes = redisTemplate.opsForSet().members(indexKey);
        if (existingHashes != null) {
            for (String hashStr : existingHashes) {
                long existingHash = Long.parseLong(hashStr);
                int distance = hammingDistance(simhash, existingHash);
                if (distance <= HAMMING_THRESHOLD) {
                    log.debug("Duplicate detected: stock={}, hammingDistance={}", stockCode, distance);
                    return true;
                }
            }
        }

        // 非重复 —— 加入索引
        redisTemplate.opsForSet().add(indexKey, String.valueOf(simhash));
        // 设置过期时间（滑窗）
        redisTemplate.expire(indexKey, Duration.ofHours(SLIDING_WINDOW_HOURS));
        // 控制索引大小，超过上限时删除旧的（近似处理）
        Long size = redisTemplate.opsForSet().size(indexKey);
        if (size != null && size > MAX_SIMHASH_INDEX) {
            // 简化处理：重建索引时丢弃旧数据，实际可按时间分桶
            log.warn("SimHash index overflow for {}, size={}, resetting", stockCode, size);
            redisTemplate.delete(indexKey);
            redisTemplate.opsForSet().add(indexKey, String.valueOf(simhash));
            redisTemplate.expire(indexKey, Duration.ofHours(SLIDING_WINDOW_HOURS));
        }
        return false;
    }

    /**
     * 汉明距离（popcount of XOR）
     */
    private int hammingDistance(long a, long b) {
        return Long.bitCount(a ^ b);
    }

    /**
     * MurmurHash3 简化实现 → 返回64位long
     */
    private long murmurHash3Long(String key) {
        // 使用Guava Hashing.murmur3_128 取前64位
        return Hashing.murmur3_128().hashString(key, StandardCharsets.UTF_8).asLong();
    }

    /**
     * 简单分词：中文二元组 + 英文空格
     */
    private List<String> tokenize(String text) {
        List<String> tokens = new ArrayList<>();
        // 英文按空格
        String[] words = text.split("\\s+");
        for (String word : words) {
            if (word.length() <= 1) continue;
            if (word.matches("[a-zA-Z0-9]+")) {
                tokens.add(word.toLowerCase());
            } else {
                // 中文二元组
                for (int i = 0; i < word.length() - 1; i++) {
                    tokens.add(word.substring(i, i + 2));
                }
            }
        }
        return tokens;
    }
}
```

```java
/**
 * AC自动机实现（精简版，~120行）
 */
class ACTrie {

    private ACTrieNode root;

    public ACTrie(List<ACTriePattern> patterns) {
        root = new ACTrieNode();
        // 1. 构建Trie
        for (ACTriePattern p : patterns) {
            insert(p.getText(), p.getTag());
        }
        // 2. 构建failure指针（BFS）
        buildFailureLinks();
    }

    private void insert(String text, String tag) {
        ACTrieNode node = root;
        for (char c : text.toCharArray()) {
            node = node.children.computeIfAbsent(c, k -> new ACTrieNode());
        }
        node.isEnd = true;
        node.tag = tag;
        node.pattern = text;
    }

    private void buildFailureLinks() {
        Queue<ACTrieNode> queue = new LinkedList<>();
        root.fail = root;
        for (ACTrieNode child : root.children.values()) {
            child.fail = root;
            queue.offer(child);
        }
        while (!queue.isEmpty()) {
            ACTrieNode current = queue.poll();
            for (Map.Entry<Character, ACTrieNode> entry : current.children.entrySet()) {
                char c = entry.getKey();
                ACTrieNode child = entry.getValue();
                ACTrieNode fail = current.fail;
                while (fail != root && !fail.children.containsKey(c)) {
                    fail = fail.fail;
                }
                child.fail = fail.children.getOrDefault(c, root);
                if (child.fail == child) child.fail = root;
                queue.offer(child);
            }
        }
    }

    /**
     * 在文本中搜索所有匹配
     */
    public List<ACTrieMatch> search(String text) {
        List<ACTrieMatch> matches = new ArrayList<>();
        ACTrieNode node = root;
        for (int i = 0; i < text.length(); i++) {
            char c = text.charAt(i);
            while (node != root && !node.children.containsKey(c)) {
                node = node.fail;
            }
            node = node.children.getOrDefault(c, root);
            // 检查当前节点及其fail链是否有匹配
            ACTrieNode temp = node;
            while (temp != root) {
                if (temp.isEnd) {
                    int start = i - temp.pattern.length() + 1;
                    matches.add(new ACTrieMatch(temp.pattern, temp.tag, start, i + 1));
                }
                temp = temp.fail;
            }
        }
        return matches;
    }
}

class ACTrieNode {
    Map<Character, ACTrieNode> children = new HashMap<>();
    ACTrieNode fail;
    boolean isEnd = false;
    String tag;
    String pattern;
}

class ACTriePattern {
    String text;
    String tag;
    // constructor, getters
}

class ACTrieMatch {

    int end;
```

```java
    /**
     * 质量评估主方法
     */
    public QualityScore evaluate(NormalizedDoc doc) {
        QualityScore score = new QualityScore();
        score.setDocId(doc.getId());

        double qualityScore = 100.0; // 基础分100，扣分制
        boolean isSpam = false;

            }
        }
        if (spamHitCount > 0) {
            qualityScore -= 50 * Math.min(spamHitCount, 2);
            isSpam = true;
            flags.add("SPAM_KEYWORD");

        // 3. 作者发帖频率检测（Redis统计24h内发帖数）

        String postCountStr = redisTemplate.opsForValue().get(authorPostKey);

        if (postCount > REPOST_THRESHOLD) {
            qualityScore -= 20;
            isSpam = true;
            flags.add("HIGH_FREQUENCY_POST");
        }
        // 更新发帖计数
        redisTemplate.opsForValue().increment(authorPostKey);
        redisTemplate.expire(authorPostKey, Duration.ofHours(24));

        // 4. 重复内容检测（与SimHash结果联动）
        if (doc.getIsDuplicate() != null && doc.getIsDuplicate()) {
            qualityScore -= 40;
            isSpam = true;
            flags.add("DUPLICATE_CONTENT");
        }

        // 5. 互动数据加权（有互动的帖子质量更高）
        int viewCount = doc.getViewCount() != null ? doc.getViewCount() : 0;
        int replyCount = doc.getReplyCount() != null ? doc.getReplyCount() : 0;
        if (viewCount > 100 || replyCount > 10) {
            qualityScore += 10; // 高互动加10分
        }

        // 6. KOL影响力评分
        double kolScore = computeKolScore(author, viewCount, replyCount);

        // 最终评分
        qualityScore = Math.max(0, Math.min(100, qualityScore));
        score.setQualityScore(qualityScore);
        score.setIsSpam(isSpam);
        score.setFlags(String.join(",", flags));
        score.setKolScore(kolScore);
        score.setEvaluatedAt(LocalDateTime.now());

        return score;
    }

    /**
     * KOL影响力评分 —— 0~100
     * 综合考虑：历史发帖量、平均互动、粉丝数（如有）、内容质量
     */
    private double computeKolScore(String author, int viewCount, int replyCount) {
        if (author == null || author.isEmpty()) return 0;

        // 从Redis获取作者历史统计
        String statsKey = "kol:stats:" + author;
        String statsStr = redisTemplate.opsForValue().get(statsKey);
        AuthorStats stats;
        if (statsStr != null) {
            stats = AuthorStats.fromJson(statsStr);
        } else {
            stats = new AuthorStats();
        }

        // 更新滚动统计
        stats.totalPosts++;
        stats.totalViews += viewCount;
        stats.totalReplies += replyCount;
        redisTemplate.opsForValue().set(statsKey, stats.toJson(), Duration.ofDays(90));

        // KOL得分公式
        double avgViews = stats.totalPosts > 0 ? (double) stats.totalViews / stats.totalPosts : 0;
        double avgReplies = stats.totalPosts > 0 ? (double) stats.totalReplies / stats.totalPosts : 0;

        // 对数缩放避免极端值
        double viewScore = Math.min(40, Math.log10(avgViews + 1) * 10);
        double replyScore = Math.min(30, Math.log10(avgReplies + 1) * 15);
        double postScore = Math.min(20, Math.log10(stats.totalPosts + 1) * 10);
        double consistencyScore = Math.min(10, stats.totalPosts * 0.1);

        return viewScore + replyScore + postScore + consistencyScore;
    }
}
```

### 3.2.5 治理流水线编排伪代码

```java
/**
 * 治理流水线编排 —— 同步Service调用替代Kafka
 * 流程：NormalizedDoc → Clean → Dedup → Entity → Quality → IndexSink
 * 每一步同步调用，失败则记录并跳过（不阻塞整批）
 */
@Service
@Slf4j
public class GovernancePipelineService {

    @Autowired private CleanWorker cleanWorker;
    @Autowired private DedupWorker dedupWorker;
    @Autowired private EntityWorker entityWorker;
    @Autowired private QualityWorker qualityWorker;
    @Autowired private DocPostMapper docPostMapper;
    @Autowired private DocEntityMentionMapper entityMentionMapper;
    @Autowired private DocQualityScoreMapper qualityScoreMapper;
    @Autowired private SentimentEngineService sentimentEngine; // 直接调用情感分析

    /** 本地阻塞队列作为采集→治理的缓冲 */
    private final BlockingQueue<List<NormalizedDoc>> pipelineQueue =
        new LinkedBlockingQueue<>(5000); // 容量5000批

    /** 治理工作线程 */
    private final ExecutorService governanceExecutor =
        Executors.newFixedThreadPool(4, r -> {
            Thread t = new Thread(r, "governance-worker");
            t.setDaemon(true);
            return t;
        });

    @PostConstruct
    public void start() {
        // 启动4个消费线程
        for (int i = 0; i < 4; i++) {
            governanceExecutor.submit(this::consumeLoop);
        }
        log.info("GovernancePipeline started with 4 workers");
    }

    /**
     * 采集器调用此方法投递文档批次
     */
    public void processBatch(List<NormalizedDoc> docs) {
        if (docs == null || docs.isEmpty()) return;
        boolean offered = pipelineQueue.offer(docs);
        if (!offered) {
            log.warn("Governance queue full, dropping {} docs", docs.size());
            // 降级：同步直接处理（背压）
            processInternal(docs);
        }
    }

    /**
     * 消费循环
     */
    private void consumeLoop() {
        while (!Thread.currentThread().isInterrupted()) {
            try {
                List<NormalizedDoc> batch = pipelineQueue.poll(5, TimeUnit.SECONDS);
                if (batch != null) {
                    processInternal(batch);
                }
            } catch (InterruptedException e) {
                Thread.currentThread().interrupt();
                break;
            } catch (Exception e) {
                log.error("Governance worker error", e);
            }
        }
    }

    /**
     * 核心处理逻辑 —— 同步顺序执行各Worker
     */
    @Transactional
    public void processInternal(List<NormalizedDoc> docs) {
        for (NormalizedDoc doc : docs) {
            try {
                // Step 1: 清洗
                doc = cleanWorker.clean(doc);

                // Step 2: 去重
                if (dedupWorker.isDuplicate(doc)) {
                    doc.setIsDuplicate(true);
                    continue; // 丢弃重复内容
                }

                // Step 3: 实体抽取
                List<EntityMention> mentions = entityWorker.extract(doc);

                // Step 4: 质量评估
                QualityScore qualityScore = qualityWorker.evaluate(doc);

                // Step 5: 落库（按文档类型写入不同表）
                persistDoc(doc, mentions, qualityScore);

                // Step 6: 直接调用情感分析引擎（同步，替代Kafka投递）
                if (!qualityScore.getIsSpam() && qualityScore.getQualityScore() > 20) {
                    sentimentEngine.analyze(doc, qualityScore);
                }

            } catch (Exception e) {
                log.error("Governance failed for doc: {}", doc.getUrl(), e);
            }
        }
    }

    /**
     * 持久化文档 + 实体提及 + 质量分
     */
    private void persistDoc(NormalizedDoc doc, List<EntityMention> mentions, QualityScore quality) {
        // 按dataType写入不同表
        switch (doc.getDataType()) {
            case "GUBA" -> {
                DocPost post = new DocPost();
                post.setSourceCode(doc.getSourceCode());
                post.setStockCode(doc.getStockCode());
                post.setTitle(doc.getTitle());
                post.setContent(doc.getContent());
                post.setAuthor(doc.getAuthor());
                post.setPublishTime(doc.getPublishTime());
                post.setViewCount(doc.getViewCount());
                post.setReplyCount(doc.getReplyCount());
                post.setUrl(doc.getUrl());
                post.setSimHash(doc.getSimHash());
                post.setQualityScore(quality.getQualityScore());
                post.setKolScore(quality.getKolScore());
                docPostMapper.insert(post);
                doc.setId(post.getId()); // 回写ID
            }
            case "NEWS" -> {
                // 写入 doc_news 表，逻辑类似
                // docNewsMapper.insert(...);
            }
            case "ANNOUNCEMENT" -> {
                // 写入 doc_announcement 表
            }
            default -> log.debug("Unsupported dataType: {}", doc.getDataType());
        }

        // 保存实体提及
        for (EntityMention mention : mentions) {
            mention.setDocId(doc.getId());
            mention.setDocType(doc.getDataType());
            entityMentionMapper.insert(mention);
        }

        // 保存质量分
        quality.setDocId(doc.getId());
        qualityScoreMapper.insert(quality);
    }

    @PreDestroy
    public void shutdown() {
        governanceExecutor.shutdown();
        try {
            governanceExecutor.awaitTermination(30, TimeUnit.SECONDS);
        } catch (InterruptedException e) {
            governanceExecutor.shutdownNow();
        }
    }
```

```java
/**
 * L1规则引擎 —— 基于金融情感词典的规则匹配
 * 功能：金融情感词典 + 否定词 + 程度副词 + 转折句处理
 * 输出：sentimentScore ∈ [-1.0, +1.0]，sentimentLabel ∈ {POSITIVE/NEUTRAL/NEGATIVE}
 */
@Component
@Slf4j
public class L1RuleEngine {

    /** 金融情感词典：词→极性分值[-1, 1] */
    private final Map<String, Double> sentimentDict = new HashMap<>();
    /** 否定词集合 */
    private final Set<String> negationWords = Set.of(
        "不", "没", "无", "非", "未", "别", "莫", "勿", "否", "不",
        "不是", "不会", "不能", "没有", "无法", "并非", "难以"
    );
    /** 程度副词：词→权重倍率 */
    private final Map<String, Double> degreeAdverbs = Map.ofEntries(
        Map.entry("非常", 1.8), Map.entry("极其", 2.0), Map.entry("特别", 1.6),
        Map.entry("十分", 1.5), Map.entry("极度", 2.0), Map.entry("尤为", 1.5),
        Map.entry("格外", 1.5), Map.entry("显著", 1.4), Map.entry("大幅", 1.5),
        Map.entry("大幅", 1.5), Map.entry("明显", 1.3), Map.entry("较为", 1.2),
        Map.entry("稍微", 0.6), Map.entry("略微", 0.5), Map.entry("些许", 0.5),
        Map.entry("一点", 0.4), Map.entry("有点", 0.7)
    );
    /** 转折词集合 */
    private final Set<String> transitionWords = Set.of(
        "但是", "不过", "然而", "可是", "虽然", "尽管", "只是", "只是",
        "实则", "实际上", "问题是", "遗憾的是"
    );

    @PostConstruct
    public void init() {
        // 金融领域正向情感词
        String[] positiveWords = {
            "利好", "大涨", "涨停", "暴涨", "强势", "突破", "创新高", "超预期",
            "增长", "盈利", "丰收", "牛股", "龙头", "加仓", "增持", "回购",
            "分红", "送转", "业绩大增", "营收增长", "净利大增", "订单饱满",
            "需求旺盛", "供不应求", "涨价", "量价齐升", "资金涌入", "主力进场",
            "机构看好", "评级上调", "目标价上调", "买入", "强烈推荐"
        };
        // 金融领域负向情感词
        String[] negativeWords = {
            "利空", "大跌", "跌停", "暴跌", "破位", "创新低", "不及预期",
            "亏损", "下滑", "下滑", "减仓", "减持", "质押", "爆仓",
            "ST", "退市", "问询", "监管", "处罚", "违规", "商誉减值",
            "业绩暴雷", "营收下滑", "净利亏损", "订单取消", "需求疲软",
            "降价", "量价齐跌", "资金出逃", "主力出货", "机构看空",
            "评级下调", "目标价下调", "卖出", "规避", "黑天鹅", "灰犀牛"
        };

        for (String w : positiveWords) sentimentDict.put(w, 1.0);
        for (String w : negativeWords) sentimentDict.put(w, -1.0);

        log.info("L1RuleEngine initialized, dict size={}", sentimentDict.size());
    }

    /**
     * 分析文本情感
     * @return SentimentResult {score: [-1,1], label, matchedWords}
     */
    public SentimentResult analyze(String text) {
        if (text == null || text.isEmpty()) {
            return SentimentResult.neutral();
        }

        List<String> tokens = tokenize(text);
        double totalScore = 0.0;
        List<String> matchedWords = new ArrayList<>();
        List<WordScore> wordScores = new ArrayList<>();

        boolean inTransitionClause = false; // 是否在转折句后半段

        for (int i = 0; i < tokens.size(); i++) {
            String token = tokens.get(i);

            // 检测转折词 —— 转折后内容权重翻倍（转折句重点在后半段）
            if (transitionWords.contains(token)) {
                inTransitionClause = true;
                continue;
            }

            // 查情感词典
            if (sentimentDict.containsKey(token)) {
                double baseScore = sentimentDict.get(token);

                // 回看前2个token，检查否定词和程度副词
                double multiplier = 1.0;
                boolean negated = false;

                for (int j = Math.max(0, i - 2); j < i; j++) {
                    String prev = tokens.get(j);
                    if (negationWords.contains(prev)) {
                        negated = true;
                    }
                    if (degreeAdverbs.containsKey(prev)) {
                        multiplier *= degreeAdverbs.get(prev);
                    }
                }

                // 否定词反转极性
                double finalScore = negated ? -baseScore : baseScore;
                finalScore *= multiplier;

                // 转折句后半段权重×1.5
                if (inTransitionClause) {
                    finalScore *= 1.5;
                }

                totalScore += finalScore;
                matchedWords.add(token);
                wordScores.add(new WordScore(token, finalScore, negated, inTransitionClause));
            }
        }

        // 归一化到[-1, 1]：tanh压缩
        double normalizedScore = Math.tanh(totalScore / 3.0); // 除以3做缩放，tanh压缩到[-1,1]

        String label;
        if (normalizedScore > 0.15) label = "POSITIVE";
        else if (normalizedScore < -0.15) label = "NEGATIVE";
        else label = "NEUTRAL";

        SentimentResult result = new SentimentResult();
        result.setScore(normalizedScore);
        result.setLabel(label);
        result.setMatchedWords(matchedWords);
        result.setWordScores(wordScores);
        result.setEngineLevel("L1");
        result.setConfidence(Math.min(1.0, Math.abs(normalizedScore) * 2 + matchedWords.size() * 0.1));
        return result;
    }

    /**
     * 分词 —— 简化版（实际可用HanLP/ansj）
     */
    private List<String> tokenize(String text) {
        List<String> tokens = new ArrayList<>();
        // 中文按2-4字滑窗 + 英文空格分词
        // 简化实现：按字符和空格混合切分
        int maxWordLen = 4;
        for (int i = 0; i < text.length(); i++) {
            if (Character.isWhitespace(text.charAt(i))) {
                continue;
            }
            // 尝试最长匹配
            boolean matched = false;
            for (int len = Math.min(maxWordLen, text.length() - i); len >= 2; len--) {
                String candidate = text.substring(i, i + len);
                if (sentimentDict.containsKey(candidate)
                    || negationWords.contains(candidate)
                    || degreeAdverbs.containsKey(candidate)
                    || transitionWords.contains(candidate)) {
                    tokens.add(candidate);
                    i += len - 1;
                    matched = true;
                    break;
                }
            }
            if (!matched) {
                tokens.add(String.valueOf(text.charAt(i)));
            }
        }
        return tokens;
    }
}
```

### 3.3.2 BudgetRouter路由策略伪代码

```java
/**
 * BudgetRouter —— 情感分析预算路由
 * 按文档价值分、当日预算、队列积压决定走L1/L2/L3
 * 当前阶段：L1全量实时，L2/L3暂缓但保留路由框架
 */
@Component
@Slf4j
public class BudgetRouter {

    @Autowired private RedisTemplate<String, String> redisTemplate;

    /** 每日L3预算（调用次数上限） */
    private static final int DAILY_L3_BUDGET = 200;
    /** 每日L2预算 */
    private static final int DAILY_L2_BUDGET = 2000;
    /** 队列积压阈值 */
    private static final int BACKLOG_THRESHOLD = 500;
    /** L3价值分阈值 */
    private static final double L3_VALUE_THRESHOLD = 70.0;
    /** L2价值分阈值 */
    private static final double L2_VALUE_THRESHOLD = 40.0;

    @Autowired(required = false) private L1RuleEngine l1Engine;
    @Autowired(required = false) private L2FinBertEngine l2Engine; // 暂缓，可能为null
    @Autowired(required = false) private L3LLMEngine l3Engine;     // 暂缓，可能为null

    /**
     * 路由决策 + 执行
     * @param doc 文档
     * @param qualityScore 质量分（含kolScore）
     */
    public SentimentResult route(NormalizedDoc doc, QualityScore qualityScore) {
        int engineLevel = decideLevel(doc, qualityScore);
        SentimentResult result;

        switch (engineLevel) {
            case 3:
                result = executeL3(doc);
                if (result == null) {
                    // L3降级到L2
                    result = executeL2(doc);
                    if (result == null) result = executeL1(doc);
                }
                break;
            case 2:
                result = executeL2(doc);
                if (result == null) result = executeL1(doc);
                break;
            default:
                result = executeL1(doc);
        }

        result.setEngineLevel("L" + engineLevel);
        return result;
    }

    /**
     * 路由决策逻辑
     */
    private int decideLevel(NormalizedDoc doc, QualityScore quality) {
        // 1. 计算文档价值分（0~100）
        double valueScore = computeValueScore(doc, quality);

        // 2. 检查队列积压
        int backlog = getQueueBacklog();
        if (backlog > BACKLOG_THRESHOLD) {
            // 积压严重 —— 全部走L1快速处理
            log.debug("Backlog={}, all to L1", backlog);
            return 1;
        }

        // 3. 检查L3预算
        int l3Used = getDailyUsed("L3");
        if (valueScore >= L3_VALUE_THRESHOLD && l3Used < DAILY_L3_BUDGET) {
            return 3;
        }

        // 4. 检查L2预算
        int l2Used = getDailyUsed("L2");
        if (valueScore >= L2_VALUE_THRESHOLD && l2Used < DAILY_L2_BUDGET) {
            return 2;
        }

        // 5. 默认L1
        return 1;
    }

    /**
     * 文档价值分计算 —— 综合互动数、KOL影响力、内容长度、情感词密度
     */
    private double computeValueScore(NormalizedDoc doc, QualityScore quality) {
        double score = 0;

        // 互动数据（40分满分）
        int views = doc.getViewCount() != null ? doc.getViewCount() : 0;
        int replies = doc.getReplyCount() != null ? doc.getReplyCount() : 0;
        score += Math.min(20, Math.log10(views + 1) * 5);
        score += Math.min(20, Math.log10(replies + 1) * 8);

        // KOL影响力（30分满分）
        score += quality.getKolScore() * 0.3;

        // 内容长度（10分满分）
        int len = doc.getContent() != null ? doc.getContent().length() : 0;
        score += Math.min(10, len / 100.0);

        // 质量分（20分满分）
        score += quality.getQualityScore() * 0.2;

        return Math.min(100, score);
    }

    private int getQueueBacklog() {
        // 从Redis获取当前积压量
        String val = redisTemplate.opsForValue().get("sentiment:backlog");
        return val != null ? Integer.parseInt(val) : 0;
    }

    private int getDailyUsed(String level) {
        String key = "sentiment:budget:" + level + ":" + LocalDate.now();
        String val = redisTemplate.opsForValue().get(key);
        return val != null ? Integer.parseInt(val) : 0;
    }

    private void incrementUsed(String level) {
        String key = "sentiment:budget:" + level + ":" + LocalDate.now();
        redisTemplate.opsForValue().increment(key);
        redisTemplate.expire(key, Duration.ofDays(2));
    }

    private SentimentResult executeL1(NormalizedDoc doc) {
        return l1Engine.analyze(doc.getTitle() + " " + doc.getContent());
    }

    private SentimentResult executeL2(NormalizedDoc doc) {
        if (l2Engine == null) return null; // L2暂缓
        incrementUsed("L2");
        return l2Engine.predict(doc.getContent());
    }

    private SentimentResult executeL3(NormalizedDoc doc) {
        if (l3Engine == null) return null; // L3暂缓
        incrementUsed("L3");
        return l3Engine.analyze(doc.getContent());
    }
}
```

### 3.3.3 情绪指数计算伪代码

```java
/**
 * 情绪指数计算 —— Si 公式实现
 * Si = 100 × Σ(wk × score_k) / Σ(wk)
 * wk = w_account × w_time × w_source × w_quality × log(1+互动数)
 * w_time: 半衰期24h，时间衰减因子 = 0.5^(Δt/24)
 */
@Service
@Slf4j
public class SentimentIndexCalculator {

    @Autowired private SentimentResultMapper sentimentResultMapper;
    @Autowired private SentimentDailyMapper sentimentDailyMapper;

    /** 半衰期24小时 */
    private static final double HALF_LIFE_HOURS = 24.0;
    /** 源权重 */
    private static final Map<String, Double> SOURCE_WEIGHTS = Map.of(
        "eastmoney", 1.0,
        "sina", 0.8,
        "10jqka", 0.9,
        "cninfo", 1.2  // 公告源权重更高
    );

    /**
     * 计算某股票某日的情绪指数
     * @param stockCode 股票代码
     * @param date 日期
     */
    public SentimentDaily calculateDaily(String stockCode, LocalDate date) {
        LocalDateTime dayStart = date.atStartOfDay();
        LocalDateTime dayEnd = date.plusDays(1).atStartOfDay();

        // 查询该股票当日所有情感分析结果
        List<SentimentResult> results = sentimentResultMapper.selectByStockAndDateRange(
            stockCode, dayStart, dayEnd);

        if (results.isEmpty()) {
            return createNeutralDaily(stockCode, date);
        }

        double weightedScoreSum = 0.0;
        double weightSum = 0.0;
        int positiveCount = 0, negativeCount = 0, neutralCount = 0;
        double totalInteraction = 0;

        LocalDateTime now = LocalDateTime.now();

        for (SentimentResult r : results) {
            double score = r.getScore(); // [-1, 1]

            // 1. w_account: 账号权重（KOL分 / 100）
            double wAccount = r.getKolScore() != null ? r.getKolScore() / 100.0 : 0.5;
            wAccount = Math.max(0.1, wAccount); // 最低0.1

            // 2. w_time: 时间衰减，半衰期24h
            double deltaHours = Duration.between(r.getPublishTime(), now).toHours();
            double wTime = Math.pow(0.5, deltaHours / HALF_LIFE_HOURS);

            // 3. w_source: 数据源权重
            double wSource = SOURCE_WEIGHTS.getOrDefault(r.getSourceCode(), 0.8);

            // 4. w_quality: 质量分权重
            double wQuality = r.getQualityScore() != null ? r.getQualityScore() / 100.0 : 0.5;
            wQuality = Math.max(0.1, wQuality);

            // 5. 互动数对数权重
            int interaction = (r.getViewCount() != null ? r.getViewCount() : 0)
                            + (r.getReplyCount() != null ? r.getReplyCount() : 0);
            double wInteraction = Math.log1p(interaction);

            // 综合权重
            double wk = wAccount * wTime * wSource * wQuality * wInteraction;

            weightedScoreSum += wk * score;
            weightSum += wk;
            totalInteraction += interaction;

            // 统计正/负/中性数量
            if (score > 0.15) positiveCount++;
            else if (score < -0.15) negativeCount++;
            else neutralCount++;
        }

        // Si = 100 × Σ(wk × score_k) / Σ(wk)
        double sentimentIndex = weightSum > 0
            ? 100.0 * weightedScoreSum / weightSum
            : 0.0;

        // 构建持久化对象
        SentimentDaily daily = new SentimentDaily();
        daily.setStockCode(stockCode);
        daily.setTradeDate(date);
        daily.setSentimentIndex(sentimentIndex); // [-100, 100]
        daily.setTotalPosts(results.size());
        daily.setPositiveCount(positiveCount);
        daily.setNegativeCount(negativeCount);
        daily.setNeutralCount(neutralCount);
        daily.setPositiveRatio((double) positiveCount / results.size());
        daily.setNegativeRatio((double) negativeCount / results.size());
        daily.setAvgScore(weightSum > 0 ? weightedScoreSum / weightSum : 0);
        daily.setTotalInteraction((long) totalInteraction);
        daily.setCalculatedAt(LocalDateTime.now());

        // upsert: 同一股票同一天只保留最新
        sentimentDailyMapper.upsert(daily);
        return daily;
    }

    /**
     * 批量计算多个股票的情绪指数
     */
    public void calculateDailyBatch(List<String> stockCodes, LocalDate date) {
        for (String code : stockCodes) {
            try {
                calculateDaily(code, date);
            } catch (Exception e) {
                log.error("Calculate sentiment daily failed: stock={}, date={}", code, date, e);
            }
        }
    }

    private SentimentDaily createNeutralDaily(String stockCode, LocalDate date) {

        daily.setSentimentIndex(0.0);

        daily.setPositiveCount(0);
        daily.setNegativeCount(0);
        daily.setNeutralCount(0);
        daily.setCalculatedAt(LocalDateTime.now());
        return daily;
    }
```

### 3.3.4 突增检测伪代码

```java
/**
 * 突增检测引擎 —— 三模型并行
 * 1. Z-Score：当前15min声量 vs 过去20日同时段均值，Z>3
 * 2. 环比突增：当日累计 > 昨日同期×3 且绝对值>30
 * 3. 情绪反转：1h内跌幅>40
 */
@Service
@Slf4j
public class AnomalyDetector {

    @Autowired private SentimentIntradayMapper intradayMapper;
    @Autowired private AnomalyEventMapper anomalyEventMapper;
    @Autowired private RedisTemplate<String, String> redisTemplate;

    private static final int Z_SCORE_THRESHOLD = 3;
    private static final double SURGE_RATIO_THRESHOLD = 3.0;
    private static final int SURGE_ABSOLUTE_MIN = 30;
    private static final double REVERSAL_DROP_THRESHOLD = 0.40; // 40%跌幅
    private static final int LOOKBACK_DAYS = 20;
    private static final int GRANULARITY_MINUTES = 15;

    /**
     * 执行三模型并行检测
     * @param stockCode 股票代码
     * @param currentTime 当前时间点
     */
    public List<AnomalyEvent> detect(String stockCode, LocalDateTime currentTime) {
        List<AnomalyEvent> anomalies = new ArrayList<>();

        // 并行执行三模型（简化为顺序调用，实际可用CompletableFuture）
        AnomalyEvent zScoreAnomaly = detectZScore(stockCode, currentTime);
        AnomalyEvent surgeAnomaly = detectSurgeRatio(stockCode, currentTime);
        AnomalyEvent reversalAnomaly = detectSentimentReversal(stockCode, currentTime);

        if (zScoreAnomaly != null) anomalies.add(zScoreAnomaly);
        if (surgeAnomaly != null) anomalies.add(surgeAnomaly);
        if (reversalAnomaly != null) anomalies.add(reversalAnomaly);

        // 持久化异常事件
        for (AnomalyEvent event : anomalies) {
            anomalyEventMapper.insert(event);
        }

        return anomalies;
    }

    /**
     * 模型1：Z-Score突增检测
     * 当前15min声量 vs 过去20日同时段均值，Z>3 则告警
     */
    private AnomalyEvent detectZScore(String stockCode, LocalDateTime now) {
        // 计算当前15min时间槽
        int currentSlot = getSlotIndex(now);
        int currentCount = getCurrentSlotVolume(stockCode, now);

        // 获取过去20日同时段（同一15min槽）的声量历史
        List<Integer> historicalVolumes = new ArrayList<>();
        for (int d = 1; d <= LOOKBACK_DAYS; d++) {
            LocalDateTime historicalDate = now.minusDays(d);
            int volume = getHistoricalSlotVolume(stockCode, historicalDate, currentSlot);
            if (volume > 0) historicalVolumes.add(volume);
        }

        if (historicalVolumes.size() < 5) return null; // 样本不足

        // 计算均值和标准差
        double mean = historicalVolumes.stream()
            .mapToInt(Integer::intValue).average().orElse(0);
        double variance = historicalVolumes.stream()
            .mapToDouble(v -> Math.pow(v - mean, 2))
            .average().orElse(0);
        double stdDev = Math.sqrt(variance);

        if (stdDev < 0.5) return null; // 标准差太小，不触发

        double zScore = (currentCount - mean) / stdDev;

        if (zScore > Z_SCORE_THRESHOLD) {
            return buildAnomalyEvent(stockCode, now, "Z_SCORE",
                String.format("声量突增: 当前=%d, 历史20日均值=%.1f, ZScore=%.2f",
                    currentCount, mean, zScore),
                zScore);
        }
        return null;
    }

    /**
     * 模型2：环比突增检测
     * 当日累计 > 昨日同期×3 且绝对值>30
     */
    private AnomalyEvent detectSurgeRatio(String stockCode, LocalDateTime now) {
        // 当日截至当前的累计声量
        int todayCumulative = getCumulativeVolumeToday(stockCode, now);
        // 昨日同期累计声量
        int yesterdayCumulative = getCumulativeVolume(stockCode,
            now.minusDays(1).toLocalDate(), now.toLocalTime());

        if (yesterdayCumulative < 1) return null;

        double ratio = (double) todayCumulative / yesterdayCumulative;

        if (ratio >= SURGE_RATIO_THRESHOLD && todayCumulative >= SURGE_ABSOLUTE_MIN) {
            return buildAnomalyEvent(stockCode, now, "SURGE_RATIO",
                String.format("环比突增: 今日累计=%d, 昨日同期=%d, 倍数=%.2f",
                    todayCumulative, yesterdayCumulative, ratio),
                ratio);
        }
        return null;
    }

    /**
     * 模型3：情绪反转检测
     * 1h内情绪指数跌幅>40%
     */
    private AnomalyEvent detectSentimentReversal(String stockCode, LocalDateTime now) {
        // 当前情绪指数（最近15min）
        double currentSentiment = getLatestSentimentIndex(stockCode, now);
        // 1小时前的情绪指数
        double sentiment1hAgo = getSentimentIndexAt(stockCode, now.minusHours(1));

        if (sentiment1hAgo <= 0) return null; // 无基准数据

        double dropRate = (sentiment1hAgo - currentSentiment) / Math.abs(sentiment1hAgo);

        if (dropRate > REVERSAL_DROP_THRESHOLD) {
            return buildAnomalyEvent(stockCode, now, "SENTIMENT_REVERSAL",
                String.format("情绪反转: 1h前=%.2f, 当前=%.2f, 跌幅=%.1f%%",
                    sentiment1hAgo, currentSentiment, dropRate * 100),
                dropRate);
        }
        return null;
    }

    /**
     * 构建异常事件
     */
    private AnomalyEvent buildAnomalyEvent(String stockCode, LocalDateTime time,
            String type, String description, double metricValue) {
        AnomalyEvent event = new AnomalyEvent();
        event.setStockCode(stockCode);
        event.setAnomalyType(type);
        event.setEventTime(time);
        event.setDescription(description);
        event.setMetricValue(metricValue);
        event.setSeverity(computeSeverity(type, metricValue));
        event.setStatus("NEW");
        return event;
    }

    private String computeSeverity(String type, double metricValue) {
        if ("Z_SCORE".equals(type) && metricValue > 5) return "CRITICAL";
        if ("Z_SCORE".equals(type) && metricValue > 3) return "HIGH";
        if ("SURGE_RATIO".equals(type) && metricValue > 5) return "CRITICAL";
        if ("SENTIMENT_REVERSAL".equals(type) && metricValue > 0.6) return "CRITICAL";
        return "MEDIUM";
    }

    /**
     * 获取15min时间槽索引 (0~95)
     * 例如 10:15 → slot=41 (10*4 + 1)
     */
    private int getSlotIndex(LocalDateTime time) {
        return time.getHour() * 4 + time.getMinute() / 15;
    }

    private int getCurrentSlotVolume(String stockCode, LocalDateTime now) {
        LocalDateTime slotStart = now.withMinute((now.getMinute() / 15) * 15).withSecond(0);
        LocalDateTime slotEnd = slotStart.plusMinutes(15);
        return intradayMapper.countPostsInSlot(stockCode, slotStart, slotEnd);
    }

    private int getHistoricalSlotVolume(String stockCode, LocalDateTime date, int slot) {
        LocalDateTime slotStart = date.toLocalDate().atStartOfDay().plusMinutes(slot * 15L);
        LocalDateTime slotEnd = slotStart.plusMinutes(15);
        return intradayMapper.countPostsInSlot(stockCode, slotStart, slotEnd);
    }

    private int getCumulativeVolumeToday(String stockCode, LocalDateTime now) {
        return intradayMapper.countPostsInSlot(stockCode,
            now.toLocalDate().atStartOfDay(), now);
    }

    private int getCumulativeVolume(String stockCode, LocalDate date, LocalTime endTime) {
        return intradayMapper.countPostsInSlot(stockCode,
            date.atStartOfDay(), date.atTime(endTime));
    }

    private double getLatestSentimentIndex(String stockCode, LocalDateTime now) {
        Double val = intradayMapper.selectLatestSentimentIndex(stockCode, now);
        return val != null ? val : 0.0;
    }

    private double getSentimentIndexAt(String stockCode, LocalDateTime time) {
        Double val = intradayMapper.selectSentimentIndexAt(stockCode, time);
        return val != null ? val : 0.0;
    }
}
```

```java
/**
 * 派生表更新服务 —— 更新 ana_sentiment_daily + ana_sentiment_intraday(15min粒度)

@Service
@Slf4j
public class SentimentDerivedService {

    @Autowired private SentimentResultMapper sentimentResultMapper;
    @Autowired private SentimentDailyMapper dailyMapper;

    @Autowired private SentimentIndexCalculator indexCalculator;

    /**
     * 更新日内15min粒度数据
     * 每个时间槽聚合该15min内的情感分析结果
     */
    @Transactional
    public void updateIntraday(String stockCode, LocalDateTime slotTime) {
        // 计算当前15min时间槽的起止
        LocalDateTime slotStart = slotTime.withMinute((slotTime.getMinute() / 15) * 15).withSecond(0).withNano(0);
        LocalDateTime slotEnd = slotStart.plusMinutes(15);

        // 查询该槽内的所有情感结果
        List<SentimentResult> results = sentimentResultMapper.selectByStockAndDateRange(
            stockCode, slotStart, slotEnd);

        SentimentIntraday intraday = new SentimentIntraday();
        intraday.setStockCode(stockCode);
        intraday.setSlotTime(slotStart);
        intraday.setSlotIndex(slotStart.getHour() * 4 + slotStart.getMinute() / 15);

        if (results.isEmpty()) {
            intraday.setPostCount(0);
            intraday.setSentimentIndex(0.0);
            intraday.setAvgScore(0.0);
        } else {
            // 计算加权情绪指数
            double weightedSum = 0, weightSum = 0;
            int positive = 0, negative = 0, neutral = 0;
            for (SentimentResult r : results) {
                double w = (r.getQualityScore() != null ? r.getQualityScore() / 100.0 : 0.5)
                         * Math.log1p((r.getViewCount() != null ? r.getViewCount() : 0)
                            + (r.getReplyCount() != null ? r.getReplyCount() : 0));
                weightedSum += w * r.getScore();
                weightSum += w;
                if (r.getScore() > 0.15) positive++;
                else if (r.getScore() < -0.15) negative++;
                else neutral++;
            }
            intraday.setPostCount(results.size());
            intraday.setSentimentIndex(weightSum > 0 ? 100.0 * weightedSum / weightSum : 0.0);
            intraday.setAvgScore(weightSum > 0 ? weightedSum / weightSum : 0.0);
            intraday.setPositiveCount(positive);
            intraday.setNegativeCount(negative);
            intraday.setNeutralCount(neutral);
        }
        intraday.setUpdatedAt(LocalDateTime.now());

        // upsert
        intradayMapper.upsert(intraday);
    }
```

### 3.4.1 综合评分卡伪代码

```java
/**
 * 综合评分卡 —— 5维加权评分
 * CompositeScore = 0.25×基本面 + 0.20×估值 + 0.20×资金面 + 0.20×技术面 + 0.15×舆情情绪
 * z-score标准化后加权，输出0~100 + 归因条形图
 */
@Service
@Slf4j
public class CompositeScoreService {

    @Autowired private FinIndicatorMapper finIndicatorMapper;
    @Autowired private QuoteDailyMapper quoteDailyMapper;
    @Autowired private MfMoneyflowMapper moneyflowMapper;
    @Autowired private SentimentDailyMapper sentimentDailyMapper;
    @Autowired private AnaFactorValueMapper factorValueMapper;
    @Autowired private AnaStockScoreMapper stockScoreMapper;

    /** 维度权重 */
    private static final double W_FUNDAMENTAL = 0.25;
    private static final double W_VALUATION = 0.20;
    private static final double W_CAPITAL = 0.20;
    private static final double W_TECHNICAL = 0.20;
    private static final double W_SENTIMENT = 0.15;

    /**
     * 计算个股综合评分
     */
    public StockScore calculate(String stockCode, LocalDate tradeDate) {
        // 1. 计算各维度原始得分（0~100）
        double fundamentalScore = calcFundamentalScore(stockCode, tradeDate);
        double valuationScore = calcValuationScore(stockCode, tradeDate);
        double capitalScore = calcCapitalScore(stockCode, tradeDate);
        double technicalScore = calcTechnicalScore(stockCode, tradeDate);
        double sentimentScore = calcSentimentScore(stockCode, tradeDate);

        // 2. z-score标准化（需要行业对标，此处用简化版：直接归一化到0~100）
        double zFundamental = zscoreNormalize(fundamentalScore);
        double zValuation = zscoreNormalize(valuationScore);
        double zCapital = zscoreNormalize(capitalScore);
        double zTechnical = zscoreNormalize(technicalScore);
        double zSentiment = zscoreNormalize(sentimentScore);

        // 3. 加权合成
        double compositeScore = W_FUNDAMENTAL * zFundamental
                              + W_VALUATION * zValuation
                              + W_CAPITAL * zCapital
                              + W_TECHNICAL * zTechnical
                              + W_SENTIMENT * zSentiment;

        // 映射到0~100（假设z-score已标准化到该范围）
        compositeScore = Math.max(0, Math.min(100, compositeScore));

        // 4. 构建归因分解
        List<ScoreAttribution> attributions = new ArrayList<>();
        attributions.add(new ScoreAttribution("基本面", zFundamental, W_FUNDAMENTAL,
            W_FUNDAMENTAL * zFundamental, fundamentalScore));
        attributions.add(new ScoreAttribution("估值", zValuation, W_VALUATION,
            W_VALUATION * zValuation, valuationScore));
        attributions.add(new ScoreAttribution("资金面", zCapital, W_CAPITAL,
            W_CAPITAL * zCapital, capitalScore));
        attributions.add(new ScoreAttribution("技术面", zTechnical, W_TECHNICAL,
            W_TECHNICAL * zTechnical, technicalScore));
        attributions.add(new ScoreAttribution("舆情情绪", zSentiment, W_SENTIMENT,
            W_SENTIMENT * zSentiment, sentimentScore));

        // 5. 构建评分等级
        String grade = computeGrade(compositeScore);

        // 6. 持久化
        StockScore score = new StockScore();
        score.setStockCode(stockCode);
        score.setTradeDate(tradeDate);
        score.setCompositeScore(compositeScore);
        score.setFundamentalScore(zFundamental);
        score.setValuationScore(zValuation);
        score.setCapitalScore(zCapital);
        score.setTechnicalScore(zTechnical);
        score.setSentimentScore(zSentiment);
        score.setGrade(grade);
        score.setAttributions(objectMapper.writeValueAsString(attributions));
        score.setCalculatedAt(LocalDateTime.now());

        stockScoreMapper.upsert(score);
        return score;
    }

    /**
     * z-score标准化 —— 将原始分映射到0~100
     * 简化版：假设输入已在0~100范围，做线性拉伸
     * 完整版需对标行业均值/标准差
     */
    private double zscoreNormalize(double rawScore) {
        // 简化：直接clamp到0~100
        // 完整版：
        // double z = (raw - industryMean) / industryStdDev;
        // return 50 + z * 15; // 映射到均值50，标准差15的正态区间
        return Math.max(0, Math.min(100, rawScore));
    }

    /**
     * 评级映射
     */
    private String computeGrade(double score) {
        if (score >= 85) return "A+";
        if (score >= 75) return "A";
        if (score >= 65) return "B+";
        if (score >= 55) return "B";
        if (score >= 45) return "C+";
        if (score >= 35) return "C";
        return "D";
    }

    // ===== 各维度得分计算 =====

    /**
     * 基本面得分（0~100）
     * 营收增速 + 净利增速 + 毛利率 + ROE + 资产负债率
     */
    private double calcFundamentalScore(String stockCode, LocalDate date) {
        FinIndicator fin = finIndicatorMapper.selectLatest(stockCode, date);
        if (fin == null) return 50.0; // 无数据给中性分

        double score = 0;
        // 营收增速（25分满分）
        double revGrowth = fin.getRevenueGrowth() != null ? fin.getRevenueGrowth() : 0;
        score += Math.min(25, Math.max(0, revGrowth * 2.5)); // 10%增速→25分

        // 净利增速（25分满分）
        double profitGrowth = fin.getProfitGrowth() != null ? fin.getProfitGrowth() : 0;
        score += Math.min(25, Math.max(0, profitGrowth * 2.5));

        // 毛利率（20分满分）
        double grossMargin = fin.getGrossMargin() != null ? fin.getGrossMargin() : 0;
        score += Math.min(20, Math.max(0, grossMargin * 0.4)); // 50%毛利率→20分

        // ROE（20分满分）
        double roe = fin.getRoe() != null ? fin.getRoe() : 0;
        score += Math.min(20, Math.max(0, roe * 2)); // 10% ROE → 20分

        // 资产负债率（10分满分，越低越好）
        double debtRatio = fin.getDebtRatio() != null ? fin.getDebtRatio() : 50;
        score += Math.min(10, Math.max(0, (100 - debtRatio) * 0.2)); // 50%负债率→10分

        return Math.min(100, score);
    }

    /**
     * 估值得分（0~100）
     * PE/PB/PS 与行业中位数对比
     */
    private double calcValuationScore(String stockCode, LocalDate date) {
        QuoteDaily quote = quoteDailyMapper.selectLatest(stockCode, date);
        FinIndicator fin = finIndicatorMapper.selectLatest(stockCode, date);
        if (quote == null || fin == null) return 50.0;

        double score = 0;
        // PE估值（40分）：PE在10~20之间为合理区间
        Double pe = quote.getPeTtm();
        if (pe != null) {
            if (pe <= 0) score += 0;        // 亏损
            else if (pe <= 15) score += 35; // 低估
            else if (pe <= 25) score += 30; // 合理
            else if (pe <= 40) score += 15; // 偏高
            else score += 5;                 // 高估
        }

        // PB估值（30分）：PB<1为低估
        Double pb = quote.getPb();
        if (pb != null) {
            if (pb <= 1) score += 28;
            else if (pb <= 2) score += 25;
            else if (pb <= 4) score += 15;
            else if (pb <= 8) score += 8;
            else score += 3;
        }

        // PS估值（30分）
        Double ps = quote.getPsTtm();
        if (ps != null) {
            if (ps <= 2) score += 28;
            else if (ps <= 5) score += 22;
            else if (ps <= 10) score += 12;
            else score += 5;
        }

        return Math.min(100, score);
    }

    /**
     * 资金面得分（0~100）
     * 主力净流入 + 大单占比 + 北向持股变化
     */
    private double calcCapitalScore(String stockCode, LocalDate date) {
        MfMoneyflow mf = moneyflowMapper.selectLatest(stockCode, date);
        if (mf == null) return 50.0;

        double score = 50.0; // 基准50

        // 主力净流入（±30分）
        Double mainNetInflow = mf.getMainNetInflow();
        if (mainNetInflow != null) {
            // 净流入为正加分，为负减分（按成交额比例）
            double turnover = mf.getTurnover() != null ? mf.getTurnover() : 1;
            double ratio = mainNetInflow / turnover;
            score += Math.max(-30, Math.min(30, ratio * 100));
        }

        // 大单占比（±10分）
        Double bigOrderRatio = mf.getBigOrderRatio();
        if (bigOrderRatio != null) {
            score += Math.max(-10, Math.min(10, (bigOrderRatio - 0.1) * 100));
        }

        // 北向持股变化（±10分）
        Double northChange = mf.getNorthHoldingChange();
        if (northChange != null) {
            score += Math.max(-10, Math.min(10, northChange * 10));
        }

        return Math.max(0, Math.min(100, score));
    }

    /**
     * 技术面得分（0~100）
     * MACD + KDJ + RSI + 均线排列 + 布林带位置
     */
    private double calcTechnicalScore(String stockCode, LocalDate date) {
        List<QuoteDaily> quotes = quoteDailyMapper.selectLatestN(stockCode, date, 60);
        if (quotes.size() < 30) return 50.0;

        double score = 0;
        BigDecimal latestClose = quotes.get(0).getClose();
        // 引用技术指标计算（见3.4.4）
        Map<String, Object> indicators = TechnicalIndicatorCalculator.calculateAll(quotes);

        // MACD（20分）：金叉+多头
        String macdSignal = (String) indicators.get("macdSignal");
        score += switch (macdSignal) {
            case "GOLDEN_CROSS_BULL" -> 20;
            case "GOLDEN_CROSS" -> 15;
            case "DEATH_CROSS_BEAR" -> 0;
            default -> 10;
        };

        // KDJ（20分）：超卖区买入信号加分
        double k = (double) indicators.get("kdjK");
        double d = (double) indicators.get("kdjD");
        if (k < 20 && k > d) score += 18; // 超卖金叉
        else if (k > 80) score += 5;      // 超买
        else score += 12;

        // RSI（20分）
        double rsi = (double) indicators.get("rsi6");
        if (rsi < 30) score += 18;  // 超卖
        else if (rsi > 70) score += 5;
        else score += 12;

        // 均线排列（20分）
        String maAlignment = (String) indicators.get("maAlignment");
        score += switch (maAlignment) {
            case "STRONG_BULL" -> 20; // 多头排列
            case "BULL" -> 15;
            case "STRONG_BEAR" -> 0;
            default -> 10;
        };

        // 布林带位置（20分）
        double bollPosition = (double) indicators.get("bollPosition"); // 0=下轨 1=上轨
        if (bollPosition < 0.2) score += 18; // 接近下轨，超卖
        else if (bollPosition > 0.8) score += 5;
        else score += 12;

        return Math.min(100, score);
    }
```

```java
    @Transactional
    public void updateWeight(Long factorId, double newWeight) {
        AnaFactorDef factor = factorDefMapper.selectById(factorId);
        if (factor == null) throw new IllegalArgumentException("Factor not found: " + factorId);

        // 校验同维度下权重之和=1.0
        List<AnaFactorDef> siblings = factorDefMapper.selectByCategory(factor.getCategory());
        double currentSum = siblings.stream()
            .filter(f -> !f.getId().equals(factorId))
            .mapToDouble(AnaFactorDef::getWeight)
            .sum();
        if (currentSum + newWeight > 1.01 || currentSum + newWeight < 0.99) {
            throw new IllegalArgumentException(
                String.format("Weight sum would be %.2f, must equal 1.0", currentSum + newWeight));
        }

        factor.setWeight(newWeight);
        factor.setUpdatedAt(LocalDateTime.now());
        factorDefMapper.updateById(factor);
        log.info("Factor {} weight updated to {}", factorId, newWeight);
    }

    /**
     * 计算某股票所有因子值
     */
    public List<AnaFactorValue> computeFactorValues(String stockCode, LocalDate tradeDate) {
        List<AnaFactorDef> activeFactors = factorDefMapper.selectAllActive();
        List<AnaFactorValue> values = new ArrayList<>();

        for (AnaFactorDef factor : activeFactors) {
            try {
                double rawValue = evaluateExpression(factor.getCalcExpression(), stockCode, tradeDate);
                double normalizedValue = normalizeValue(rawValue, factor);
                double weightedValue = normalizedValue * factor.getWeight();

                AnaFactorValue fv = new AnaFactorValue();

                fv.setStockCode(stockCode);
                fv.setTradeDate(tradeDate);
                fv.setRawValue(rawValue);
                fv.setNormalizedValue(normalizedValue);
                fv.setWeightedValue(weightedValue);
                fv.setCalculatedAt(LocalDateTime.now());
                values.add(fv);

                factorValueMapper.upsert(fv);
            } catch (Exception e) {
                log.error("Compute factor {} failed: {}", factor.getFactorCode(), e.getMessage());
            }
        }
        return values;
    }

            case "ROE" -> { return finIndicatorService.getRoe(stockCode, date); }
            case "GROSS_MARGIN" -> { return finIndicatorService.getGrossMargin(stockCode, date); }
            case "PE_TTM" -> { return quoteService.getPeTtm(stockCode, date); }
            case "PB" -> { return quoteService.getPb(stockCode, date); }
            case "MAIN_NET_INFLOW_RATIO" -> { return moneyFlowService.getMainInflowRatio(stockCode, date); }

            case "KDJ_SCORE" -> { return technicalService.getKdjScore(stockCode, date); }
            case "RSI_SCORE" -> { return technicalService.getRsiScore(stockCode, date); }
            case "SENTIMENT_INDEX" -> { return sentimentService.getSentimentIndex(stockCode, date); }
            default -> { return factor.getDefaultValue(); }
        }

    /**

        // DESC: 越小越好 → 反转

        }
        return Math.max(0, Math.min(100, raw + 50)); // 简化映射
    }
```

```java
/**
 * 技术指标计算 —— MACD / KDJ / RSI / 布林带 / 均线排列
 */
@Component
@Slfj4j
public class TechnicalIndicatorCalculator {

    /**
     * 计算全部技术指标
     * @param quotes 日线数据（按日期降序，最新在前）
     */
    public static Map<String, Object> calculateAll(List<QuoteDaily> quotes) {
        Map<String, Object> result = new HashMap<>();
        // 反转为升序方便计算
        List<BigDecimal> closes = quotes.stream()
            .sorted(Comparator.comparing(QuoteDaily::getTradeDate))
            .map(QuoteDaily::getClose)
            .collect(Collectors.toList());

        if (closes.size() < 30) {
            result.put("macdSignal", "INSUFFICIENT_DATA");
            return result;
        }

        double[] closeArr = closes.stream().mapToDouble(BigDecimal::doubleValue).toArray();

        // 1. MACD
        result.putAll(calcMACD(closeArr));

        // 2. KDJ
        result.putAll(calcKDJ(quotes));

        // 3. RSI
        result.putAll(calcRSI(closeArr));

        // 4. 布林带
        result.putAll(calcBollinger(closeArr));

        // 5. 均线排列
        result.putAll(calcMAAlignment(closeArr));

        return result;
    }

    /**
     * MACD计算
     * EMA12, EMA26 → DIF → DEA(EMA9 of DIF) → MACD柱 = 2×(DIF-DEA)
     */
    private static Map<String, Object> calcMACD(double[] closes) {
        double[] ema12 = calcEMA(closes, 12);
        double[] ema26 = calcEMA(closes, 26);

        // DIF = EMA12 - EMA26
        double[] dif = new double[closes.length];
        for (int i = 0; i < closes.length; i++) {
            dif[i] = ema12[i] - ema26[i];
        }

        // DEA = EMA(DIF, 9)
        double[] dea = calcEMA(dif, 9);

        // MACD柱 = 2 × (DIF - DEA)
        double[] macd = new double[closes.length];
        for (int i = 0; i < closes.length; i++) {
            macd[i] = 2 * (dif[i] - dea[i]);
        }

        int lastIdx = closes.length - 1;
        Map<String, Object> result = new HashMap<>();
        result.put("macdDif", dif[lastIdx]);
        result.put("macdDea", dea[lastIdx]);
        result.put("macdHist", macd[lastIdx]);

        // 判断信号
        if (lastIdx >= 1) {
            boolean goldenCross = dif[lastIdx] > dea[lastIdx] && dif[lastIdx - 1] <= dea[lastIdx - 1];
            boolean deathCross = dif[lastIdx] < dea[lastIdx] && dif[lastIdx - 1] >= dea[lastIdx - 1];
            boolean bullTrend = dif[lastIdx] > 0 && dea[lastIdx] > 0;
            boolean bearTrend = dif[lastIdx] < 0 && dea[lastIdx] < 0;

            if (goldenCross && bullTrend) result.put("macdSignal", "GOLDEN_CROSS_BULL");
            else if (goldenCross) result.put("macdSignal", "GOLDEN_CROSS");
            else if (deathCross && bearTrend) result.put("macdSignal", "DEATH_CROSS_BEAR");
            else if (deathCross) result.put("macdSignal", "DEATH_CROSS");
            else if (dif[lastIdx] > dea[lastIdx]) result.put("macdSignal", "BULL");
            else result.put("macdSignal", "BEAR");
        }
        return result;
    }

    /**
     * EMA计算
     */
    private static double[] calcEMA(double[] values, int period) {
        double[] ema = new double[values.length];
        double multiplier = 2.0 / (period + 1);
        ema[0] = values[0];
        for (int i = 1; i < values.length; i++) {
            ema[i] = (values[i] - ema[i - 1]) * multiplier + ema[i - 1];
        }
        return ema;
    }

    /**
     * KDJ计算
     * RSV = (Close - Low_n) / (High_n - Low_n) × 100
     * K(n) = 2/3 × K(n-1) + 1/3 × RSV
     * D(n) = 2/3 × D(n-1) + 1/3 × K(n)
     * J = 3K - 2D
     */
    private static Map<String, Object> calcKDJ(List<QuoteDaily> quotes) {
        int n = 9; // RSV周期
        int len = quotes.size();
        List<QuoteDaily> sorted = quotes.stream()
            .sorted(Comparator.comparing(QuoteDaily::getTradeDate))
            .collect(Collectors.toList());

        double[] k = new double[len];
        double[] d = new double[len];
        k[0] = 50;
        d[0] = 50;

        for (int i = n - 1; i < len; i++) {
            // 取最近n日的最高价和最低价
            double highN = Double.MIN_VALUE;
            double lowN = Double.MAX_VALUE;
            for (int j = i - n + 1; j <= i; j++) {
                highN = Math.max(highN, sorted.get(j).getHigh().doubleValue());
                lowN = Math.min(lowN, sorted.get(j).getLow().doubleValue());
            }
            double close = sorted.get(i).getClose().doubleValue();
            double rsv = highN != lowN ? (close - lowN) / (highN - lowN) * 100 : 50;

            k[i] = 2.0 / 3 * k[i - 1] + 1.0 / 3 * rsv;
            d[i] = 2.0 / 3 * d[i - 1] + 1.0 / 3 * k[i];
        }

        int last = len - 1;
        double j = 3 * k[last] - 2 * d[last];

        Map<String, Object> result = new HashMap<>();
        result.put("kdjK", k[last]);
        result.put("kdjD", d[last]);
        result.put("kdjJ", j);
        return result;
    }

    /**
     * RSI计算
     * RSI = 100 - 100/(1 + RS), RS = avgGain/avgLoss
     */
    private static Map<String, Object> calcRSI(double[] closes) {
        int[] periods = {6, 12, 24};
        Map<String, Object> result = new HashMap<>();

        for (int period : periods) {
            if (closes.length < period + 1) continue;

            double avgGain = 0;
            double avgLoss = 0;

            // 初始平均（简单平均）
            for (int i = 1; i <= period; i++) {
                double change = closes[i] - closes[i - 1];
                if (change > 0) avgGain += change;
                else avgLoss += -change;
            }
            avgGain /= period;
            avgLoss /= period;

            // 后续用 Wilder平滑法
            for (int i = period + 1; i < closes.length; i++) {
                double change = closes[i] - closes[i - 1];
                double gain = change > 0 ? change : 0;
                double loss = change < 0 ? -change : 0;
                avgGain = (avgGain * (period - 1) + gain) / period;
                avgLoss = (avgLoss * (period - 1) + loss) / period;
            }

            double rs = avgLoss == 0 ? 100 : avgGain / avgLoss;
            double rsi = 100 - 100 / (1 + rs);
            result.put("rsi" + period, rsi);
        }
        return result;
    }

    /**
     * 布林带计算
     * 中轨 = MA20, 上轨 = MA20 + 2σ, 下轨 = MA20 - 2σ
     */
    private static Map<String, Object> calcBollinger(double[] closes) {
        int period = 20;
        if (closes.length < period) return Map.of("bollPosition", 0.5);

        // MA20
        double sum = 0;
        for (int i = closes.length - period; i < closes.length; i++) {
            sum += closes[i];
        }
        double ma = sum / period;

        // 标准差
        double variance = 0;
        for (int i = closes.length - period; i < closes.length; i++) {
            variance += Math.pow(closes[i] - ma, 2);
        }
        double stdDev = Math.sqrt(variance / period);

        double upper = ma + 2 * stdDev;
        double lower = ma - 2 * stdDev;
        double current = closes[closes.length - 1];

        // 当前价格在布林带中的位置 (0=下轨, 1=上轨)
        double position = upper != lower ? (current - lower) / (upper - lower) : 0.5;
        position = Math.max(0, Math.min(1, position));

        Map<String, Object> result = new HashMap<>();
        result.put("bollUpper", upper);
        result.put("bollMiddle", ma);
        result.put("bollLower", lower);
        result.put("bollPosition", position);
        return result;
    }

    /**
     * 均线排列判断
     * MA5 / MA10 / MA20 / MA60 的排列方式
     */
    private static Map<String, Object> calcMAAlignment(double[] closes) {
        double ma5 = calcSMA(closes, 5);
        double ma10 = calcSMA(closes, 10);
        double ma20 = calcSMA(closes, 20);
        double ma60 = closes.length >= 60 ? calcSMA(closes, 60) : ma20;

        Map<String, Object> result = new HashMap<>();
        result.put("ma5", ma5);
        result.put("ma10", ma10);
        result.put("ma20", ma20);
        result.put("ma60", ma60);

        // 判断排列
        if (ma5 > ma10 && ma10 > ma20 && ma20 > ma60) {
            result.put("maAlignment", "STRONG_BULL"); // 多头排列
        } else if (ma5 > ma10 && ma10 > ma20) {
            result.put("maAlignment", "BULL");
        } else if (ma5 < ma10 && ma10 < ma20 && ma20 < ma60) {
            result.put("maAlignment", "STRONG_BEAR"); // 空头排列
        } else if (ma5 < ma10 && ma10 < ma20) {
            result.put("maAlignment", "BEAR");
        } else {
            result.put("maAlignment", "MIXED");
        }
        return result;
    }

    private static double calcSMA(double[] closes, int period) {
        if (closes.length < period) period = closes.length;
        double sum = 0;
        for (int i = closes.length - period; i < closes.length; i++) {
            sum += closes[i];
        }
        return sum / period;
    }
}
```

```java
/**

        // 单日主力净流入

            MfMoneyflow latest = flows.get(0);
            analysis.setMainNetInflow(latest.getMainNetInflow());
            analysis.setBigOrderRatio(latest.getBigOrderRatio());
            analysis.setSuperBigOrderRatio(latest.getSuperBigOrderRatio());
            analysis.setTurnover(latest.getTurnover());

            double ratio = latest.getTurnover() != null && latest.getTurnover() > 0
                ? (latest.getMainNetInflow() / latest.getTurnover()) * 100 : 0;
            analysis.setMainInflowRatio(ratio);

            // 资金趋势判断
            String trend = determineFlowTrend(flows);
            analysis.setFlowTrend(trend);
        }

        // 2. 大单占比分析
        analysis.setBigOrderNetInflow(flows.stream()
            .mapToDouble(f -> f.getBigOrderNetInflow() != null ? f.getBigOrderNetInflow() : 0)
            .sum());

        // 3. 龙虎榜分析
        List<DragonTigerEntry> dragonList = dragonTigerMapper.selectByStockAndDate(stockCode, tradeDate);
        if (!dragonList.isEmpty()) {
            analysis.setHasDragonTiger(true);
            analysis.setDragonTigerEntries(dragonList);
            // 机构席位净买入
            double orgNetBuy = dragonList.stream()
                .filter(e -> "INSTITUTIONAL".equals(e.getSeatType()))
                .mapToDouble(e -> e.getNetBuy())
                .sum();
            analysis.setOrgNetBuyDragonTiger(orgNetBuy);
        }

        // 4. 北向持股变化
        List<NorthHolding> holdings = northHoldingMapper.selectLatestN(stockCode, tradeDate, 5);
        if (holdings.size() >= 2) {
            NorthHolding latest = holdings.get(0);
            NorthHolding prev = holdings.get(1);
            double shareChange = latest.getHoldingShares() - prev.getHoldingShares();
            double valueChange = latest.getHoldingValue() - prev.getHoldingValue();
            analysis.setNorthShareChange(shareChange);
            analysis.setNorthValueChange(valueChange);
            analysis.setNorthHoldingRatio(latest.getHoldingRatio());
            // 北向趋势
            analysis.setNorthTrend(determineNorthTrend(holdings));
        }

        return analysis;
    }

    private String determineFlowTrend(List<MfMoneyflow> flows) {
        // 近5日主力净流入连续正/负
        long positiveDays = flows.stream()
            .filter(f -> f.getMainNetInflow() != null && f.getMainNetInflow() > 0)
            .count();
        if (positiveDays >= 4) return "STRONG_INFLOW";
        if (positiveDays >= 3) return "INFLOW";
        if (positiveDays <= 1) return "STRONG_OUTFLOW";
        if (positiveDays <= 2) return "OUTFLOW";
        return "BALANCED";
    }

    private String determineNorthTrend(List<NorthHolding> holdings) {
        // 判断北向最近是否在加仓
        int increasing = 0;
        for (int i = 0; i < holdings.size() - 1; i++) {
            if (holdings.get(i).getHoldingShares() > holdings.get(i + 1).getHoldingShares()) {
                increasing++;
            }
        }
        if (increasing >= 4) return "ACCUMULATING";
        if (increasing <= 1) return "REDUCING";
        return "STABLE";
    }
}
```

```java
/**
 * 基本面分析 —— 营收/净利增速/毛利率/ROE/杜邦拆解
 */
@Service
@Slf4j
public class FundamentalService {

    @Autowired private FinIndicatorMapper indicatorMapper;
    @Autowired private FinIncomeMapper incomeMapper;
    @Autowired private FinBalanceMapper balanceMapper;
    @Autowired private FinCashflowMapper cashflowMapper;

    /**
     * 基本面综合分析
     */
    public FundamentalAnalysis analyze(String stockCode, LocalDate tradeDate) {
        FundamentalAnalysis analysis = new FundamentalAnalysis();
        analysis.setStockCode(stockCode);

        // 获取最近8期财报数据
        List<FinIndicator> indicators = indicatorMapper.selectLatest8(stockCode, tradeDate);

        if (indicators.isEmpty()) {
            analysis.setHasData(false);
            return analysis;
        }

        FinIndicator latest = indicators.get(0);
        analysis.setHasData(true);
        analysis.setReportDate(latest.getReportDate());

        // 1. 营收增速（同比 + 8期趋势）
        analysis.setRevenueGrowth(latest.getRevenueGrowth());
        analysis.setRevenueGrowthTrend(extractTrend(indicators, FinIndicator::getRevenueGrowth));

        // 2. 净利增速
        analysis.setProfitGrowth(latest.getProfitGrowth());
        analysis.setProfitGrowthTrend(extractTrend(indicators, FinIndicator::getProfitGrowth));

        // 3. 毛利率
        analysis.setGrossMargin(latest.getGrossMargin());
        analysis.setGrossMarginTrend(extractTrend(indicators, FinIndicator::getGrossMargin));

        // 4. ROE
        analysis.setRoe(latest.getRoe());
        analysis.setRoeTrend(extractTrend(indicators, FinIndicator::getRoe));

        // 5. 杜邦拆解：ROE = 净利率 × 总资产周转率 × 权益乘数
        DupontAnalysis dupont = calculateDupont(latest);
        analysis.setDupont(dupont);

        // 6. 资产负债率
        analysis.setDebtRatio(latest.getDebtRatio());

        // 7. 行业排名（如果有行业数据）
        // analysis.setIndustryRank(indicatorMapper.selectIndustryRank(stockCode, tradeDate));

        // 8. 8期趋势数据（供前端图表）
        analysis.setTrendData(buildTrendData(indicators));

        return analysis;
    }

    /**
     * 杜邦分析
     * ROE = 销售净利率 × 总资产周转率 × 权益乘数
     * 销售净利率 = 净利润 / 营收
     * 总资产周转率 = 营收 / 总资产
     * 权益乘数 = 总资产 / 净资产 = 1/(1-资产负债率)
     */
    private DupontAnalysis calculateDupont(FinIndicator fin) {
        DupontAnalysis dupont = new DupontAnalysis();
        dupont.setRoe(fin.getRoe());

        // 销售净利率
        double netMargin = fin.getRevenue() != null && fin.getRevenue() > 0
            ? fin.getNetProfit() / fin.getRevenue() * 100 : 0;
        dupont.setNetProfitMargin(netMargin);

        // 总资产周转率
        double assetTurnover = fin.getTotalAssets() != null && fin.getTotalAssets() > 0
            ? fin.getRevenue() / fin.getTotalAssets() : 0;
        dupont.setAssetTurnover(assetTurnover);

        // 权益乘数
        double equityMultiplier = fin.getDebtRatio() != null
            ? 100.0 / (100.0 - fin.getDebtRatio()) : 1.0;
        dupont.setEquityMultiplier(equityMultiplier);

        // 验证：ROE ≈ 净利率 × 周转率 × 权益乘数
        double calculatedRoe = (netMargin / 100) * assetTurnover * equityMultiplier * 100;
        dupont.setCalculatedRoe(calculatedRoe);

        return dupont;
    }

    /**
     * 提取8期趋势方向
     */
    private String extractTrend(List<FinIndicator> indicators,
            Function<FinIndicator, Double> getter) {
        if (indicators.size() < 3) return "INSUFFICIENT";
        // indicators按时间降序（最新在前），反转后升序
        List<Double> values = indicators.stream()
            .sorted(Comparator.comparing(FinIndicator::getReportDate))
            .map(getter)
            .filter(Objects::nonNull)
            .collect(Collectors.toList());

        if (values.size() < 3) return "INSUFFICIENT";

        // 判断趋势：线性回归斜率
        int n = values.size();
        double sumX = 0, sumY = 0, sumXY = 0, sumX2 = 0;
        for (int i = 0; i < n; i++) {
            sumX += i;
            sumY += values.get(i);
            sumXY += i * values.get(i);
            sumX2 += (double) i * i;
        }
        double slope = (n * sumXY - sumX * sumY) / (n * sumX2 - sumX * sumX);

        if (slope > 0.5) return "UPWARD";
        if (slope < -0.5) return "DOWNWARD";
        return "STABLE";
    }

    /**
     * 构建8期趋势数据（供前端图表）
     */
    private List<TrendDataPoint> buildTrendData(List<FinIndicator> indicators) {
        return indicators.stream()
            .sorted(Comparator.comparing(FinIndicator::getReportDate))
            .map(fin -> {
                TrendDataPoint point = new TrendDataPoint();
                point.setReportDate(fin.getReportDate());
                point.setRevenue(fin.getRevenue());
                point.setNetProfit(fin.getNetProfit());
                point.setRevenueGrowth(fin.getRevenueGrowth());
                point.setProfitGrowth(fin.getProfitGrowth());
                point.setGrossMargin(fin.getGrossMargin());
                point.setRoe(fin.getRoe());
                return point;
            })
            .collect(Collectors.toList());
    }
}
```

### 3.5.1 规则DSL解析伪代码

```java
/**
 * 告警规则DSL解析与执行引擎
 * JSON DSL → 条件树 → 执行引擎
 *
 * DSL示例:
 * {
 *   "name":"情绪突降",
 *   "scope":"watchlist",
 *   "condition":{"and":[
 *     {"metric":"sentiment_index_1h","op":"drop","value":40},
 *     {"metric":"volume_ratio","op":">","value":1.5}
 *   ]},
 *   "cooldown":"4h","maxPerDay":3,"channels":["webpush","email"]
 * }
 */
@Service
@Slf4j
public class AlertRuleEngine {

    @Autowired private AlertRuleMapper ruleMapper;
    @Autowired private AlertEventMapper eventMapper;
    @Autowired private ObjectMapper objectMapper;
    @Autowired private MetricQueryService metricQueryService;
    @Autowired private NoiseReducer noiseReducer;
    @Autowired private AlertNotifier alertNotifier;

    /**
     * 加载并解析规则DSL
     */
    public AlertRule parse(String dslJson) throws Exception {
        JsonNode dsl = objectMapper.readTree(dslJson);
        AlertRule rule = new AlertRule();
        rule.setName(dsl.path("name").asText());
        rule.setScope(dsl.path("scope").asText());     // watchlist / single / system
        rule.setStockCode(dsl.path("stockCode").asText(null));
        rule.setDsl(dslJson);

        // 解析cooldown
        String cooldown = dsl.path("cooldown").asText("4h");
        rule.setCooldownSeconds(parseDuration(cooldown));

        // 解析maxPerDay
        rule.setMaxPerDay(dsl.path("maxPerDay").asInt(3));

        // 解析channels
        List<String> channels = new ArrayList<>();
        JsonNode channelsNode = dsl.path("channels");
        if (channelsNode.isArray()) {
            for (JsonNode ch : channelsNode) channels.add(ch.asText());
        }
        rule.setChannels(channels);

        // 解析条件树
        JsonNode conditionNode = dsl.path("condition");
        ConditionNode conditionTree = parseConditionNode(conditionNode);
        rule.setConditionTree(conditionTree);

        rule.setStatus(1);
        rule.setCreatedAt(LocalDateTime.now());
        return rule;
    }

    /**
     * 递归解析条件树
     */
    private ConditionNode parseConditionNode(JsonNode node) {
        if (node.has("and")) {
            AndNode and = new AndNode();
            for (JsonNode child : node.path("and")) {
                and.addCondition(parseConditionNode(child));
            }
            return and;
        }
        if (node.has("or")) {
            OrNode or = new OrNode();
            for (JsonNode child : node.path("or")) {
                or.addCondition(parseConditionNode(child));
            }
            return or;
        }
        // 叶子条件
        LeafNode leaf = new LeafNode();
        leaf.setMetric(node.path("metric").asText());
        leaf.setOp(node.path("op").asText());    // >, <, =, >=, <=, drop, surge
        leaf.setValue(node.path("value").asDouble());
        return leaf;
    }

    /**
     * 执行规则检查
     * @param rule 告警规则
     * @param stockCode 被检查的股票
     * @return 触发则返回AlertEvent，否则null
     */
    public AlertEvent evaluate(AlertRule rule, String stockCode) {
        // 1. 检查冷却时间
        if (!noiseReducer.checkCooldown(rule, stockCode)) {
            return null; // 在冷却期内
        }

        // 2. 检查日上限
        if (!noiseReducer.checkDailyLimit(rule, stockCode)) {
            return null; // 超过当日上限
        }

        // 3. 检查夜间勿扰
        if (!noiseReducer.checkNightMode(rule)) {
            return null;
        }

        // 4. 执行条件树求值
        boolean triggered = evaluateConditionTree(rule.getConditionTree(), stockCode);
        if (!triggered) return null;

        // 5. 构建告警事件
        AlertEvent event = new AlertEvent();
        event.setRuleId(rule.getId());
        event.setRuleName(rule.getName());
        event.setStockCode(stockCode);
        event.setTriggerTime(LocalDateTime.now());
        event.setScope(rule.getScope());
        event.setChannels(String.join(",", rule.getChannels()));
        event.setStatus("NEW");
        event.setPriority(determinePriority(rule));

        // 6. 降噪处理
        event = noiseReducer.mergeOrPass(event);

        // 7. 持久化
        eventMapper.insert(event);

        // 8. 触发通知
        alertNotifier.notify(event, rule.getChannels());

        return event;
    }

    /**
     * 递归执行条件树求值
     */
    private boolean evaluateConditionTree(ConditionNode node, String stockCode) {
        if (node instanceof AndNode andNode) {
            // AND: 所有子条件都满足
            return andNode.getConditions().stream()
                .allMatch(child -> evaluateConditionTree(child, stockCode));
        }
        if (node instanceof OrNode orNode) {
            // OR: 任一子条件满足
            return orNode.getConditions().stream()
                .anyMatch(child -> evaluateConditionTree(child, stockCode));
        }
        if (node instanceof LeafNode leaf) {
            return evaluateLeaf(leaf, stockCode);
        }
        return false;
    }

    /**
     * 叶子条件求值 —— 查询指标并比较
     */
    private boolean evaluateLeaf(LeafNode leaf, String stockCode) {
        // 查询当前指标值
        double currentValue = metricQueryService.queryMetric(leaf.getMetric(), stockCode);

        return switch (leaf.getOp()) {
            case ">" -> currentValue > leaf.getValue();
            case "<" -> currentValue < leaf.getValue();
            case ">=" -> currentValue >= leaf.getValue();
            case "<=" -> currentValue <= leaf.getValue();
            case "=" -> Math.abs(currentValue - leaf.getValue()) < 0.01;
            case "drop" -> {
                // 跌幅超过value%
                double previousValue = metricQueryService.queryPreviousMetric(leaf.getMetric(), stockCode);
                if (previousValue == 0) yield false;
                double dropRate = (previousValue - currentValue) / Math.abs(previousValue) * 100;
                yield dropRate > leaf.getValue();
            }
            case "surge" -> {
                // 涨幅超过value%
                double previousValue = metricQueryService.queryPreviousMetric(leaf.getMetric(), stockCode);
                if (previousValue == 0) yield false;
                double surgeRate = (currentValue - previousValue) / Math.abs(previousValue) * 100;
                yield surgeRate > leaf.getValue();
            }
            default -> false;
        };
    }

    /**
     * 确定优先级
     */
    private String determinePriority(AlertRule rule) {
        // P0: 价格异动/资金异动
        // P1: 舆情情绪变化
        // P2: 日报/统计
        if (rule.getName().contains("异动") || rule.getName().contains("暴跌") || rule.getName().contains("涨停")) {
            return "P0";
        }
        if (rule.getName().contains("情绪") || rule.getName().contains("舆情")) {
            return "P1";
        }
        return "P2";
    }

    private long parseDuration(String duration) {
        if (duration.endsWith("h")) return Long.parseLong(duration.substring(0, duration.length() - 1)) * 3600;
        if (duration.endsWith("m")) return Long.parseLong(duration.substring(0, duration.length() - 1)) * 60;
        if (duration.endsWith("s")) return Long.parseLong(duration.substring(0, duration.length() - 1));
        return 4 * 3600; // 默认4h
    }
}

/**
 * 条件树节点接口
 */
interface ConditionNode {}

class AndNode implements ConditionNode {
    private final List<ConditionNode> conditions = new ArrayList<>();
    void addCondition(ConditionNode c) { conditions.add(c); }
    List<ConditionNode> getConditions() { return conditions; }
}

class OrNode implements ConditionNode {
    private final List<ConditionNode> conditions = new ArrayList<>();
    void addCondition(ConditionNode c) { conditions.add(c); }
    List<ConditionNode> getConditions() { return conditions; }
}

class LeafNode implements ConditionNode {
    private String metric;
    private String op;
    private double value;
    // getters/setters
}
```

### 3.5.2 降噪机制伪代码

```java
/**
 * 降噪机制 —— 告警合并与抑制
 * 1. 同类事件15min窗口合并
 * 2. 同一标的日上限
 * 3. 夜间勿扰（22:00-08:00 仅P0告警）
 * 4. 优先级队列排序
 */
@Service
@Slf4j
public class NoiseReducer {

    @Autowired private RedisTemplate<String, String> redisTemplate;
    @Autowired private AlertEventMapper eventMapper;

    private static final int MERGE_WINDOW_MINUTES = 15;
    private static final int NIGHT_START_HOUR = 22;
    private static final int NIGHT_END_HOUR = 8;

    /**
     * 检查冷却时间
     * @return true=可以告警, false=冷却中
     */
    public boolean checkCooldown(AlertRule rule, String stockCode) {
        String key = String.format("alert:cooldown:%d:%s", rule.getId(), stockCode);
        String lastTrigger = redisTemplate.opsForValue().get(key);
        if (lastTrigger != null) {
            log.debug("Alert in cooldown: rule={}, stock={}", rule.getId(), stockCode);
            return false;
        }
        // 设置冷却
        redisTemplate.opsForValue().set(key, String.valueOf(System.currentTimeMillis()),
            Duration.ofSeconds(rule.getCooldownSeconds()));
        return true;
    }

    /**
     * 检查当日上限
     */
    public boolean checkDailyLimit(AlertRule rule, String stockCode) {
        String key = String.format("alert:daily:%d:%s:%s",
            rule.getId(), stockCode, LocalDate.now());
        String count = redisTemplate.opsForValue().get(key);
        int currentCount = count != null ? Integer.parseInt(count) : 0;
        if (currentCount >= rule.getMaxPerDay()) {
            log.debug("Alert daily limit reached: rule={}, stock={}, count={}",
                rule.getId(), stockCode, currentCount);
            return false;
        }
        return true;
    }

    /**
     * 检查夜间勿扰
     */
    public boolean checkNightMode(AlertRule rule) {
        int hour = LocalTime.now().getHour();
        boolean isNight = hour >= NIGHT_START_HOUR || hour < NIGHT_END_HOUR;
        if (isNight) {
            // 夜间仅P0级别告警通过
            String priority = determinePriority(rule);
            if (!"P0".equals(priority)) {
                log.debug("Night mode, non-P0 alert suppressed: {}", rule.getName());
                return false;
            }
        }
        return true;
    }

    /**
     * 同类事件15min窗口合并
     * 如果同一规则+同一股票在15min内已触发过，合并为一条
     */
    public AlertEvent mergeOrPass(AlertEvent event) {
        String mergeKey = String.format("alert:merge:%d:%s",
            event.getRuleId(), event.getStockCode());

        // 检查15min内是否已有同类事件
        String existingId = redisTemplate.opsForValue().get(mergeKey);
        if (existingId != null) {
            // 合并到已有事件
            AlertEvent existing = eventMapper.selectById(Long.parseLong(existingId));
            if (existing != null) {
                existing.setMergeCount(existing.getMergeCount() + 1);
                existing.setLastTriggerTime(LocalDateTime.now());
                eventMapper.updateById(existing);
                log.info("Alert merged: rule={}, stock={}, mergeCount={}",
                    event.getRuleId(), event.getStockCode(), existing.getMergeCount());
                return existing; // 返回已合并的事件，不再新增
            }
        }

        // 新事件 —— 设置合并窗口
        redisTemplate.opsForValue().set(mergeKey, String.valueOf(event.getId()),
            Duration.ofMinutes(MERGE_WINDOW_MINUTES));

        // 增加当日计数
        String dailyKey = String.format("alert:daily:%d:%s:%s",
            event.getRuleId(), event.getStockCode(), LocalDate.now());
        redisTemplate.opsForValue().increment(dailyKey);
        redisTemplate.expire(dailyKey, Duration.ofDays(2));

        event.setMergeCount(1);
        return event;
    }

    /**
     * 优先级队列排序 —— P0 > P1 > P2
     */
    public List<AlertEvent> sortByPriority(List<AlertEvent> events) {
        return events.stream()
            .sorted(Comparator
                .comparing(AlertEvent::getPriority, (a, b) -> {
                    Map<String, Integer> order = Map.of("P0", 0, "P1", 1, "P2", 2);
                    return order.getOrDefault(a, 3) - order.getOrDefault(b, 3);
                })
                .thenComparing(AlertEvent::getTriggerTime, Comparator.reverseOrder()))
            .collect(Collectors.toList());
    }

    private String determinePriority(AlertRule rule) {
        if (rule.getName().contains("异动") || rule.getName().contains("暴跌")) return "P0";
        if (rule.getName().contains("情绪")) return "P1";
        return "P2";
    }
}
```

```java
        HttpEntity<Map<String, Object>> entity = new HttpEntity<>(payload, headers);

            channel.getWebhookUrl(), entity, String.class);

        if (!response.getStatusCode().is2xxSuccessful()) {
            throw new RuntimeException("DingTalk webhook failed: " + response.getStatusCode());
        }
    }

        return String.format("[%s] %s - %s", event.getPriority(), event.getRuleName(), event.getStockCode());
    }

    private String formatContent(AlertEvent event) {
        return String.format("触发时间: %s\n规则: %s\n股票: %s\n触发值: %s",
            event.getTriggerTime(), event.getRuleName(), event.getStockCode(),
            event.getTriggerValues());
    }

    private void saveNotifyLog(Long eventId, String channel, String status, String error) {
        AlertNotifyLog log = new AlertNotifyLog();
        log.setEventId(eventId);
        log.setChannel(channel);
        log.setStatus(status);
        log.setErrorMsg(error);
        log.setSentAt(LocalDateTime.now());
```

```typescript
/**
 * 计算移动平均
 */
function computeMA(data: number[], period: number): number[] {
  const result: number[] = []
  for (let i = 0; i < data.length; i++) {
    if (i < period - 1) {
      result.push(NaN)
      continue
    }
    let sum = 0
    for (let j = 0; j < period; j++) sum += data[i - j]
    result.push(sum / period)
  }
  return result
}
```

```typescript
import { ref, reactive, computed } from 'vue'

const props = defineProps<{ ruleId?: number }>()
const emit = defineEmits<{ save: [dsl: string] }>()

const isEdit = computed(() => !!props.ruleId)

interface ConditionRow {
  metric: string
  op: string
  value: number
}

const formRef = ref()
const conditionLogic = ref<'and' | 'or'>('and')
const conditions = ref<ConditionRow[]>([{ metric: 'sentiment_index_1h', op: 'drop', value: 40 }])
const stockOptions = ref<{ code: string; name: string }[]>([])

  name: '',
  scope: 'watchlist',
  stockCode: '',
  cooldown: '4h',
  maxPerDay: 3,
  nightMode: true,
  channels: ['webpush']

  channels: [{ required: true, type: 'array', min: 1, message: '至少选择一个通知渠道', trigger: 'change' }]
}

// ===== 生成DSL JSON =====
const generatedDSL = computed(() => {
  const conditionObj: any = {}
  const logicKey = conditionLogic.value // 'and' or 'or'
  conditionObj[logicKey] = conditions.value.map(c => ({
    metric: c.metric,
    op: c.op,
    value: c.value
  }))

  const dsl: any = {
    name: form.name || '未命名规则',
    scope: form.scope
  }

  if (form.scope === 'single' && form.stockCode) {
    dsl.stockCode = form.stockCode
  }

  dsl.condition = conditionObj
  dsl.cooldown = form.cooldown
  dsl.maxPerDay = form.maxPerDay
  dsl.nightMode = form.nightMode
  dsl.channels = form.channels

  return JSON.stringify(dsl, null, 2)
})

// ===== 操作方法 =====
function addCondition() {
  conditions.value.push({ metric: 'volume_ratio', op: '>', value: 1.5 })
}

function removeCondition(idx: number) {
  conditions.value.splice(idx, 1)
}

function getMetricUnit(metric: string): string {
  const units: Record<string, string> = {
    sentiment_index_1h: '分',

    volume_ratio: '倍',
    price_change_pct: '%',
    main_net_inflow: '万',
    composite_score: '分',
    sentiment_reversal: '%'

    if (isEdit.value) {
      await alertRuleApi.update(props.ruleId!, generatedDSL.value)
    } else {
      await alertRuleApi.create(generatedDSL.value)

}

}
```

```java
@Service
@Slf4j
public class StockDetailAggregator {

    @Autowired private QuoteService quoteService;
    @Autowired private SentimentService sentimentService;
    @Autowired private TechnicalService technicalService;
    @Autowired private CapitalFlowService capitalFlowService;
    @Autowired private FundamentalService fundamentalService;
    @Autowired private ResearchService researchService;
    @Autowired private RiskService riskService;
    @Autowired private CompositeScoreService scoreService;

    private final ExecutorService aggregatorExecutor =
        Executors.newFixedThreadPool(7, r -> {
            Thread t = new Thread(r, "stock-detail-agg");
            t.setDaemon(true);
            return t;
        });

    /**
     * 并行聚合所有卡片数据
     */
    public StockDetailVO aggregate(String stockCode) {
        StockDetailVO vo = new StockDetailVO();
        vo.setStockCode(stockCode);

        // 并行调用7个子服务
        CompletableFuture<Void> f1 = CompletableFuture.runAsync(
            () -> vo.setOverview(getOverview(stockCode)), aggregatorExecutor);
        CompletableFuture<Void> f2 = CompletableFuture.runAsync(
            () -> vo.setSentiment(getSentiment(stockCode, 30)), aggregatorExecutor);
        CompletableFuture<Void> f3 = CompletableFuture.runAsync(
            () -> vo.setTechnical(getTechnical(stockCode, "daily", 60)), aggregatorExecutor);
        CompletableFuture<Void> f4 = CompletableFuture.runAsync(
            () -> vo.setCapital(getCapital(stockCode, 5)), aggregatorExecutor);
        CompletableFuture<Void> f5 = CompletableFuture.runAsync(
            () -> vo.setFundamental(getFundamental(stockCode)), aggregatorExecutor);
        CompletableFuture<Void> f6 = CompletableFuture.runAsync(
            () -> vo.setResearch(getResearch(stockCode)), aggregatorExecutor);
        CompletableFuture<Void> f7 = CompletableFuture.runAsync(
            () -> vo.setRisk(getRisk(stockCode)), aggregatorExecutor);

        // 等待全部完成（超时3秒）
        try {
            CompletableFuture.allOf(f1, f2, f3, f4, f5, f6, f7)
                .get(3, TimeUnit.SECONDS);
        } catch (TimeoutException e) {
            log.warn("Stock detail aggregation timeout for {}, partial data", stockCode);
        } catch (Exception e) {
            log.error("Stock detail aggregation failed for {}", stockCode, e);
        }

        return vo;
    }

        return List.of(
            quotes.stream().mapToDouble(q -> q.getHigh().doubleValue()).max().orElse(0),
```

```java
@Component
@Slf4j
public class WatchlistWebSocketHandler implements WebSocketHandler {

    @Autowired private RedisTemplate<String, String> redisTemplate;
    @Autowired private ObjectMapper objectMapper;

    /** userId → WebSocket会话集合 */
    private final ConcurrentMap<Long, Set<WebSocketSession>> userSessions = new ConcurrentHashMap<>();

    @Override
    public void afterConnectionEstablished(WebSocketSession session) {
        Long userId = (Long) session.getAttributes().get("userId");
        userSessions.computeIfAbsent(userId, k -> ConcurrentHashMap.newKeySet()).add(session);
        log.info("WebSocket connected: userId={}, sessionId={}", userId, session.getId());
    }

    @Override
    public void handleMessage(WebSocketSession session, WebSocketMessage<?> message) {
        // 处理客户端心跳
        if (message.getPayload().toString().equals("ping")) {
            try {
                session.sendMessage(new TextMessage("pong"));
            } catch (IOException e) {
                log.error("Send pong failed", e);
            }
        }
    }

    @Override
    public void afterConnectionClosed(WebSocketSession session, CloseStatus status) {
        Long userId = (Long) session.getAttributes().get("userId");
        Set<WebSocketSession> sessions = userSessions.get(userId);
        if (sessions != null) {
            sessions.remove(session);
            if (sessions.isEmpty()) userSessions.remove(userId);
        }
        log.info("WebSocket closed: userId={}, sessionId={}", userId, session.getId());
    }

    /**
     * 推送自选股变更到用户所有在线设备
     */
    public void pushWatchlistChange(Long userId, String type, String stockCode) {
        Set<WebSocketSession> sessions = userSessions.get(userId);
        if (sessions == null || sessions.isEmpty()) return;

        Map<String, Object> payload = Map.of(
            "type", "watchlist_change",
            "action", type,        // ADD / REMOVE / SORT
            "stockCode", stockCode,
            "timestamp", System.currentTimeMillis()
        );

        String json;
        try {
            json = objectMapper.writeValueAsString(payload);
        } catch (Exception e) {
            log.error("Serialize watchlist change failed", e);
            return;
        }

        for (WebSocketSession session : sessions) {
            if (session.isOpen()) {
                try {
                    session.sendMessage(new TextMessage(json));
                } catch (IOException e) {
                    log.error("Push watchlist change failed: userId={}", userId, e);
                }
            }
        }
    }

    /**
     * Redis写扩散 —— 自选股中的股票有异动时，推送到所有持有该股的用户
     */
    @Scheduled(fixedDelay = 5000) // 每5秒扫描一次推送队列
    public void consumePushQueue() {
        // 从Redis队列拉取待推送消息
        String queueKey = "watchlist:push:queue";
        List<String> messages = redisTemplate.opsForList().range(queueKey, 0, 49); // 批量取50条
        if (messages == null || messages.isEmpty()) return;

        for (String msg : messages) {
            try {
                PushMessage pm = objectMapper.readValue(msg, PushMessage.class);
                // 查询所有持有该股票的用户
                List<Long> userIds = watchlistService.getUserIdsByStockCode(pm.getStockCode());
                for (Long userId : userIds) {
                    pushWatchlistChange(userId, pm.getType(), pm.getStockCode());
                }
            } catch (Exception e) {
                log.error("Consume push queue failed: {}", msg, e);
            }
        }
        // 移除已处理的消息
        redisTemplate.opsForList().trim(queueKey, messages.size(), -1);
    }

    @Override
    public void handleTransportError(WebSocketSession session, Throwable exception) {
        log.error("WebSocket transport error: {}", session.getId(), exception);
    }

    @Override
    public boolean supportsPartialMessages() { return false; }
}
```
